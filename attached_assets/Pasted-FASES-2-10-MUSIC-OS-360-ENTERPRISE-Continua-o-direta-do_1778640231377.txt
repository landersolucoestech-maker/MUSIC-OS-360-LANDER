2.2 — ClerkAuthGuard
typescript// apps/api/src/core/guards/clerk-auth.guard.ts
import {
  Injectable, CanActivate, ExecutionContext,
  UnauthorizedException, Logger,
} from '@nestjs/common';
import { createClerkClient } from '@clerk/backend';
import { ConfigService } from '@nestjs/config';
import { Reflector } from '@nestjs/core';

export const IS_PUBLIC_KEY = 'isPublic';

@Injectable()
export class ClerkAuthGuard implements CanActivate {
  private readonly clerk;
  private readonly logger = new Logger(ClerkAuthGuard.name);

  constructor(
    private config: ConfigService,
    private reflector: Reflector,
  ) {
    this.clerk = createClerkClient({
      secretKey: this.config.get<string>('CLERK_SECRET_KEY'),
    });
  }

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const isPublic = this.reflector.getAllAndOverride<boolean>(IS_PUBLIC_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);
    if (isPublic) return true;

    const request = context.switchToHttp().getRequest();
    const token = this.extractToken(request);

    if (!token) {
      throw new UnauthorizedException('Token de autenticação ausente');
    }

    try {
      const payload = await this.clerk.verifyToken(token);

      request.auth = {
        userId:   payload.sub,
        sessionId: payload.sid,
        orgId:    payload.org_id ?? null,
        orgRole:  payload.org_role ?? null,
        claims:   payload,
      };

      return true;
    } catch (err) {
      this.logger.warn(`Token inválido: ${(err as Error).message}`);
      throw new UnauthorizedException('Token inválido ou expirado');
    }
  }

  private extractToken(request: any): string | null {
    const auth = request.headers['authorization'];
    if (auth?.startsWith('Bearer ')) return auth.slice(7);
    return request.cookies?.['__session'] ?? null;
  }
}

2.3 — TenantGuard
typescript// apps/api/src/core/guards/tenant.guard.ts
import {
  Injectable, CanActivate, ExecutionContext,
  UnauthorizedException, ForbiddenException, Logger,
} from '@nestjs/common';
import { PrismaService } from '../database/prisma.service';

@Injectable()
export class TenantGuard implements CanActivate {
  private readonly logger = new Logger(TenantGuard.name);

  constructor(private prisma: PrismaService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const auth = request.auth;

    if (!auth?.orgId) {
      throw new UnauthorizedException('Organização não identificada no token');
    }

    const tenant = await this.prisma.tenant.findUnique({
      where: { clerk_org_id: auth.orgId },
      include: { organization: true },
    });

    if (!tenant || tenant.deleted_at) {
      throw new UnauthorizedException('Tenant não encontrado ou inativo');
    }

    const member = await this.prisma.orgMember.findUnique({
      where: {
        tenant_id_clerk_user_id: {
          tenant_id: tenant.id,
          clerk_user_id: auth.userId,
        },
      },
    });

    if (!member || !member.is_active) {
      throw new ForbiddenException('Usuário não é membro ativo deste tenant');
    }

    request.tenant        = tenant;
    request.currentMember = member;

    return true;
  }
}

2.5 — RolesGuard
typescript// apps/api/src/core/guards/roles.guard.ts
import {
  Injectable, CanActivate, ExecutionContext, ForbiddenException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { ROLES_KEY } from '../decorators/roles.decorator';

const ROLE_HIERARCHY: Record<string, number> = {
  viewer:  1,
  editor:  2,
  manager: 3,
  admin:   4,
  owner:   5,
};

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const required = this.reflector.getAllAndOverride<string[]>(ROLES_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);
    if (!required || required.length === 0) return true;

    const { currentMember } = context.switchToHttp().getRequest();
    const memberLevel = ROLE_HIERARCHY[currentMember?.role ?? 'viewer'] ?? 0;
    const minRequired  = Math.min(...required.map(r => ROLE_HIERARCHY[r] ?? 99));

    if (memberLevel < minRequired) {
      throw new ForbiddenException(
        `Permissão insuficiente. Necessário: ${required.join(' ou ')}`,
      );
    }
    return true;
  }
}

2.6 — Webhook Clerk (sync automático de org e usuários)
typescript// apps/api/src/modules/auth/clerk-webhook.controller.ts
import {
  Controller, Post, Headers, Req,
  BadRequestException, Logger,
} from '@nestjs/common';
import { Webhook } from 'svix';
import { ConfigService } from '@nestjs/config';
import { Public } from '../../core/decorators/public.decorator';
import { ClerkSyncService } from './clerk-sync.service';
import { RawBodyRequest } from '@nestjs/common';

@Controller('webhooks/clerk')
export class ClerkWebhookController {
  private readonly logger = new Logger(ClerkWebhookController.name);

  constructor(
    private config: ConfigService,
    private clerkSync: ClerkSyncService,
  ) {}

  @Post()
  @Public()
  async handle(
    @Headers('svix-id') svixId: string,
    @Headers('svix-timestamp') svixTimestamp: string,
    @Headers('svix-signature') svixSignature: string,
    @Req() req: RawBodyRequest<Request>,
  ) {
    const secret = this.config.get<string>('CLERK_WEBHOOK_SECRET')!;
    const wh = new Webhook(secret);

    let event: any;
    try {
      event = wh.verify(req.rawBody as any, {
        'svix-id':        svixId,
        'svix-timestamp': svixTimestamp,
        'svix-signature': svixSignature,
      });
    } catch {
      throw new BadRequestException('Assinatura do webhook Clerk inválida');
    }

    this.logger.log(`Clerk webhook recebido: ${event.type}`);
    await this.clerkSync.process(event);

    return { received: true };
  }
}

// apps/api/src/modules/auth/clerk-sync.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../../core/database/prisma.service';
import { DEFAULT_FEATURE_FLAGS } from '../../shared/constants/feature-flags';

@Injectable()
export class ClerkSyncService {
  private readonly logger = new Logger(ClerkSyncService.name);

  constructor(private prisma: PrismaService) {}

  async process(event: any) {
    switch (event.type) {
      case 'organization.created':
        await this.onOrgCreated(event.data);
        break;
      case 'organization.updated':
        await this.onOrgUpdated(event.data);
        break;
      case 'organizationMembership.created':
        await this.onMemberAdded(event.data);
        break;
      case 'organizationMembership.deleted':
        await this.onMemberRemoved(event.data);
        break;
      case 'user.deleted':
        await this.onUserDeleted(event.data);
        break;
      default:
        this.logger.debug(`Evento Clerk ignorado: ${event.type}`);
    }
  }

  private async onOrgCreated(data: any) {
    await this.prisma.$transaction(async (tx) => {
      // 1. Criar organização
      const org = await tx.organization.upsert({
        where:  { clerk_org_id: data.id },
        create: {
          clerk_org_id: data.id,
          name:         data.name,
          slug:         data.slug,
          industry:     'gravadora',
          plan:         'starter',
        },
        update: { name: data.name, slug: data.slug },
      });

      // 2. Criar tenant vinculado
      await tx.tenant.upsert({
        where:  { clerk_org_id: data.id },
        create: {
          org_id:      org.id,
          clerk_org_id: data.id,
          name:         data.name,
          slug:         data.slug,
          plan:         'starter',
          features:     DEFAULT_FEATURE_FLAGS,
        },
        update: {},
      });

      // 3. Criar subscription billing (trial)
      await tx.billingSubscription.upsert({
        where:  { stripe_customer_id: `pending_${org.id}` },
        create: {
          org_id:             org.id,
          stripe_customer_id: `pending_${org.id}`,
          plan:               'starter',
          status:             'trial',
          trial_ends_at:      new Date(Date.now() + 14 * 86400000),
          seats:              3,
        },
        update: {},
      });
    });

    this.logger.log(`Org criada e sincronizada: ${data.name}`);
  }

  private async onOrgUpdated(data: any) {
    await this.prisma.organization.updateMany({
      where: { clerk_org_id: data.id },
      data:  { name: data.name },
    });
  }

  private async onMemberAdded(data: any) {
    const tenant = await this.prisma.tenant.findUnique({
      where: { clerk_org_id: data.organization.id },
    });
    if (!tenant) return;

    await this.prisma.orgMember.upsert({
      where: {
        tenant_id_clerk_user_id: {
          tenant_id:     tenant.id,
          clerk_user_id: data.public_user_data.user_id,
        },
      },
      create: {
        org_id:        tenant.org_id,
        tenant_id:     tenant.id,
        clerk_user_id: data.public_user_data.user_id,
        email:         data.public_user_data.identifier ?? '',
        full_name:     `${data.public_user_data.first_name ?? ''} ${data.public_user_data.last_name ?? ''}`.trim(),
        role:          data.role === 'org:admin' ? 'admin' : 'viewer',
        joined_at:     new Date(),
        is_active:     true,
      },
      update: {
        role:      data.role === 'org:admin' ? 'admin' : 'viewer',
        is_active: true,
      },
    });
  }

  private async onMemberRemoved(data: any) {
    const tenant = await this.prisma.tenant.findUnique({
      where: { clerk_org_id: data.organization.id },
    });
    if (!tenant) return;

    await this.prisma.orgMember.updateMany({
      where: {
        tenant_id:     tenant.id,
        clerk_user_id: data.public_user_data.user_id,
      },
      data: { is_active: false },
    });
  }

  private async onUserDeleted(data: any) {
    await this.prisma.orgMember.updateMany({
      where: { clerk_user_id: data.id },
      data:  { is_active: false },
    });
  }
}

// SUBSTITUIR implementação atual por bridge com Clerk
// mantendo a mesma interface para não quebrar nenhum componente

import {
  useAuth as useClerkAuth,
  useUser,
  useOrganization,
} from '@clerk/clerk-react';
import React, { createContext, useContext } from 'react';
import type { AuthError, Session, User } from '@/shared/types/auth';

interface AuthContextType {
  user:            User | null;
  session:         Session | null;
  loading:         boolean;
  signIn:          (email: string, password: string) => Promise<{ error: AuthError | null }>;
  signUp:          (email: string, password: string, fullName?: string) => Promise<{ error: AuthError | null }>;
  signOut:         () => Promise<void>;
  resetPassword:   (email: string) => Promise<{ error: AuthError | null }>;
  updatePassword:  (password: string) => Promise<{ error: AuthError | null }>;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const { isLoaded, isSignedIn, signOut: clerkSignOut, getToken } = useClerkAuth();
  const { user: clerkUser } = useUser();
  const { organization } = useOrganization();

  const user: User | null = clerkUser
    ? {
        id:    clerkUser.id,
        email: clerkUser.primaryEmailAddress?.emailAddress ?? '',
        user_metadata: {
          full_name:  clerkUser.fullName ?? '',
          role:       organization?.membership?.role ?? 'viewer',
          org_id:     organization?.id ?? '',
          avatar_url: clerkUser.imageUrl,
        },
      }
    : null;

  const session: Session | null = isSignedIn
    ? { access_token: '', user: user! }
    : null;

  // Injetar token real nos headers de API
  // O api-client do frontend precisa buscar o token via getToken()
  React.useEffect(() => {
    if (!isSignedIn) return;
    getToken().then((token) => {
      if (token) {
        // setAccessToken já existe em shared/lib/api-client.ts
        import('@/shared/lib/api-client').then(({ setAccessToken }) => {
          setAccessToken(token);
        });
      }
    });
  }, [isSignedIn, getToken]);

  const value: AuthContextType = {
    user,
    session,
    loading: !isLoaded,
    signIn:          async () => ({ error: null }), // Clerk usa UI própria
    signUp:          async () => ({ error: null }), // Clerk usa UI própria
    signOut:         async () => { await clerkSignOut(); },
    resetPassword:   async () => ({ error: null }),
    updatePassword:  async () => ({ error: null }),
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth deve ser usado dentro de AuthProvider');
  return ctx;
}
