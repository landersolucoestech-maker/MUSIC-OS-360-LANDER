import { Test, TestingModule } from '@nestjs/testing';
import { ArtistsService }   from './artists.service';
import { PrismaService }    from '../../core/database/prisma.service';
import { EncryptionService } from '../../core/security/encryption.service';

describe('ArtistsService', () => {
  let service: ArtistsService;
  let prisma:  PrismaService;

  const mockPrisma = {
    forTenant: jest.fn().mockReturnValue({
      artist: {
        findMany:  jest.fn(),
        findFirst: jest.fn(),
        create:    jest.fn(),
        update:    jest.fn(),
        count:     jest.fn(),
      },
    }),
  };

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        ArtistsService,
        { provide: PrismaService,    useValue: mockPrisma },
        { provide: EncryptionService, useValue: {
          encrypt:         (v: string) => `enc:${v}`,
          decrypt:         (v: string) => v.replace('enc:', ''),
          encryptNullable: (v: string | null) => v ? `enc:${v}` : null,
          decryptNullable: (v: string | null) => v ? v.replace('enc:', '') : null,
        }},
      ],
    }).compile();

    service = module.get<ArtistsService>(ArtistsService);
    prisma  = module.get<PrismaService>(PrismaService);
  });

  it('deve criptografar CPF ao criar artista', async () => {
    const db = mockPrisma.forTenant('tenant-1');
    (db.artist.create as jest.Mock).mockResolvedValue({
      id: 'art-1', nome_artistico: 'Teste', cpf_cnpj_encrypted: 'enc:123',
    });

    const result = await service.create('tenant-1', 'user-1', {
      nome_artistico: 'Teste', tipo: 'solo', cpf_cnpj: '123.456.789-01',
    } as any);

    const createCall = (db.artist.create as jest.Mock).mock.calls[0][0];
    expect(createCall.data.cpf_cnpj_encrypted).toBe('enc:123.456.789-01');
    expect(createCall.data.cpf_cnpj).toBeUndefined();
  });

  it('deve descriptografar CPF ao retornar artista', async () => {
    const db = mockPrisma.forTenant('tenant-1');
    (db.artist.findFirst as jest.Mock).mockResolvedValue({
      id: 'art-1', nome_artistico: 'Teste',
      cpf_cnpj_encrypted:  'enc:123.456.789-01',
      email_encrypted:     null,
      telefone_encrypted:  null,
      manager_contato_encrypted: null,
    });

    const result = await service.findById('tenant-1', 'art-1');
    expect(result.cpf_cnpj).toBe('123.456.789-01');
    expect(result.cpf_cnpj_encrypted).toBeUndefined();
  });
});

describe('Tenant Isolation (E2E)', () => {
  it('IDOR: usuário do tenant A não acessa dados do tenant B', async () => {
    // Criar artista no tenant A
    const artistA = await request(app.getHttpServer())
      .post('/api/v1/artists')
      .set('Authorization', `Bearer ${tokenTenantA}`)
      .send({ nome_artistico: 'Artista A', tipo: 'solo' })
      .expect(201);

    // Tenant B tenta acessar artista do tenant A
    await request(app.getHttpServer())
      .get(`/api/v1/artists/${artistA.body.data.id}`)
      .set('Authorization', `Bearer ${tokenTenantB}`)
      .expect(404); // Deve retornar 404, não 403 (não revelar existência)
  });

  it('Rate limiting: 6ª requisição de auth retorna 429', async () => {
    for (let i = 0; i < 5; i++) {
      await request(app.getHttpServer())
        .post('/api/v1/auth/login')
        .send({ email: 'test@test.com', password: 'wrong' });
    }
    await request(app.getHttpServer())
      .post('/api/v1/auth/login')
      .send({ email: 'test@test.com', password: 'wrong' })
      .expect(429);
  });
});
