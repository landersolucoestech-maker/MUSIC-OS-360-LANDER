Sem Docker. Desenvolvimento direto contra serviços cloud.

CONTEXTO ABSOLUTO DO PROJETO
O MUSIC OS 360° é um ERP Musical SaaS multi-tenant atualmente implementado como frontend Vite + React puro, 100% em mock mode (localStorage + dados seed em memória). Possui ~107K linhas TypeScript, 22 módulos de negócio, arquitetura modular sólida, 9 tabelas marcadas como PENDING sem backend, todas as integrações simuladas, zero persistência real, zero autenticação real, zero backend funcional.
O objetivo é transformar isso em plataforma SaaS enterprise real.
DECISÕES ARQUITETURAIS IMUTÁVEIS:

Frontend: Vite + React permanece como está — estrutura, UX, visual e módulos preservados integralmente
A flag VITE_USE_MOCK=true mantém tudo funcionando enquanto o backend é construído fase a fase
A transição mock → real acontece módulo por módulo, nunca tudo de uma vez
Sem Docker em nenhuma etapa — desenvolvimento direto contra serviços cloud
Sem Next.js — Vite permanece para sempre neste escopo

CONFIGURAÇÃO DE DESENVOLVIMENTO SEM DOCKER
Todo o stack de desenvolvimento usa serviços cloud diretamente:

1. BANCO DE DADOS (Neon):
   - Criar projeto no Neon: https://neon.tech
   - Criar branch "dev" separado da main
   - DATABASE_URL=postgresql://... (branch dev do Neon)
   - DATABASE_DIRECT_URL=postgresql://... (para migrations Prisma)
   - Neon fornece connection pooling nativo — não precisa de nada local

2. CACHE E RATE LIMITING (Upstash):
   - Criar database Redis no Upstash: https://upstash.com
   - Usar REST API do Upstash (funciona em qualquer ambiente sem instalação)
   - UPSTASH_REDIS_URL=https://...
   - UPSTASH_REDIS_TOKEN=...

3. FILAS BULLMQ (Railway Redis):
   - Criar serviço Redis no Railway: https://railway.app
   - REDIS_QUEUE_URL=redis://...
   - BullMQ conecta diretamente ao Railway Redis em dev

4. STORAGE (Cloudflare R2):
   - Criar bucket "musicos360-dev" no R2
   - Separado do bucket de produção
   - R2_BUCKET_NAME=musicos360-dev

5. AUTH (Clerk):
   - Criar application em modo development no Clerk
   - Chaves de test nunca enviam emails reais

6. EMAIL (Resend):
   - Usar domínio de test do Resend em dev
   - Emails ficam no log, não são entregues

RESULTADO: `npm run dev` no backend conecta direto a todos os serviços cloud.
Zero instalação local além de Node.js e npm.

typescriptimport { z } from 'zod';

const envSchema = z.object({
  NODE_ENV: z.enum(['development', 'staging', 'production', 'test'])
    .default('development'),
  PORT: z.coerce.number().default(3001),

  // Neon PostgreSQL
  DATABASE_URL: z.string()
    .url()
    .refine(url => url.includes('neon.tech') || url.includes('localhost') || process.env.NODE_ENV === 'test',
      'Use Neon PostgreSQL. Sem banco local.'),
  DATABASE_DIRECT_URL: z.string().url().optional(),

  // Redis
  REDIS_QUEUE_URL: z.string().min(1, 'REDIS_QUEUE_URL obrigatório (Railway Redis)'),
  UPSTASH_REDIS_URL: z.string().url().optional(),
  UPSTASH_REDIS_TOKEN: z.string().optional(),

  // Auth
  CLERK_SECRET_KEY: z.string().startsWith('sk_', 'CLERK_SECRET_KEY inválida'),
  CLERK_WEBHOOK_SECRET: z.string().startsWith('whsec_'),

  // CORS
  CORS_ORIGINS: z.string().default('http://localhost:5000'),

  // Criptografia
  ENCRYPTION_KEY: z.string()
    .length(64, 'ENCRYPTION_KEY deve ter exatamente 64 chars hex (32 bytes = 256 bits)'),
  ENCRYPTION_IV_SECRET: z.string().min(32),

  // Stripe
  STRIPE_SECRET_KEY: z.string().optional(),
  STRIPE_WEBHOOK_SECRET: z.string().optional(),

  // Cloudflare R2
  R2_ACCOUNT_ID: z.string().optional(),
  R2_ACCESS_KEY: z.string().optional(),
  R2_SECRET_KEY: z.string().optional(),
  R2_BUCKET_NAME: z.string().optional(),
  R2_PUBLIC_URL: z.string().optional(),

  // IA
  OPENAI_API_KEY: z.string().optional(),
  ANTHROPIC_API_KEY: z.string().optional(),
  GOOGLE_AI_API_KEY: z.string().optional(),

  // Email
  RESEND_API_KEY: z.string().optional(),
  RESEND_FROM_EMAIL: z.string().email().default('noreply@musicos360.com.br'),

  // Monitoramento
  SENTRY_DSN: z.string().url().optional(),
  POSTHOG_API_KEY: z.string().optional(),
});

export type EnvConfig = z.infer<typeof envSchema>;

export function validateEnv(config: Record<string, unknown>): EnvConfig {
  const result = envSchema.safeParse(config);
  if (!result.success) {
    console.error('\n❌ ERRO: Variáveis de ambiente inválidas ou ausentes:\n');
    result.error.issues.forEach(issue => {
      console.error(`  ${issue.path.join('.')}: ${issue.message}`);
    });
    console.error('\nVerifique seu apps/api/.env\n');
    process.exit(1);
  }
  return result.data;
}

typescriptimport { NestFactory } from '@nestjs/core';
import { ValidationPipe, Logger } from '@nestjs/common';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import helmet from 'helmet';
import compression from 'compression';
import { AppModule } from './app.module';
import { GlobalExceptionFilter } from './core/filters/global-exception.filter';
import { TransformInterceptor } from './core/interceptors/transform.interceptor';
import { LoggingInterceptor } from './core/interceptors/logging.interceptor';

async function bootstrap() {
  const logger = new Logger('Bootstrap');

  const app = await NestFactory.create(AppModule, {
    bufferLogs: true,
  });

  app.use(helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        styleSrc: ["'self'", "'unsafe-inline'"],
        imgSrc: ["'self'", 'data:', 'https:'],
        scriptSrc: ["'self'"],
      },
    },
    crossOriginEmbedderPolicy: false,
  }));

  app.use(compression());

  app.enableCors({
    origin: (origin, callback) => {
      const allowed = (process.env.CORS_ORIGINS ?? 'http://localhost:5000').split(',');
      if (!origin || allowed.includes(origin)) {
        callback(null, true);
      } else {
        callback(new Error(`CORS bloqueado: ${origin}`));
      }
    },
    credentials: true,
    methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
    allowedHeaders: ['Content-Type', 'Authorization', 'X-Tenant-ID', 'X-Request-ID'],
  });

  app.setGlobalPrefix('api/v1');

  app.useGlobalPipes(new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
  }));

  app.useGlobalFilters(new GlobalExceptionFilter());
  app.useGlobalInterceptors(new TransformInterceptor(), new LoggingInterceptor());

  // Swagger apenas em dev e staging
  if (process.env.NODE_ENV !== 'production') {
    const config = new DocumentBuilder()
      .setTitle('MUSIC OS 360° API')
      .setDescription('Enterprise Music Management SaaS API')
      .setVersion('1.0')
      .addBearerAuth()
      .build();
    SwaggerModule.setup('docs', app, SwaggerModule.createDocument(app, config), {
      swaggerOptions: { persistAuthorization: true },
    });
  }

  app.enableShutdownHooks();

  const port = process.env.PORT ?? 3001;
  await app.listen(port);
  logger.log(`🎵 MUSIC OS 360° API → http://localhost:${port}`);
  logger.log(`📚 Swagger → http://localhost:${port}/docs`);
}

bootstrap().catch(err => {
  console.error('Falha crítica no bootstrap:', err);
  process.exit(1);
});

model Organization {
  id             String    @id @default(uuid())
  clerk_org_id   String    @unique
  name           String
  slug           String    @unique
  plan           String    @default("starter")
  billing_status String    @default("trial")
  industry       String    @default("gravadora")
  cnpj_encrypted String?
  phone          String?
  address        Json?
  config         Json?
  metadata       Json?
  created_at     DateTime  @default(now())
  updated_at     DateTime  @updatedAt
  deleted_at     DateTime?

  tenant        Tenant?
  members       OrgMember[]
  subscriptions BillingSubscription[]

  @@map("organizations")
}

model Tenant {
  id           String    @id @default(uuid())
  org_id       String    @unique
  clerk_org_id String    @unique
  name         String
  slug         String    @unique
  plan         String    @default("starter")
  features     Json
  settings     Json?
  created_at   DateTime  @default(now())
  updated_at   DateTime  @updatedAt
  deleted_at   DateTime?

  organization Organization @relation(fields: [org_id], references: [id])
  members      OrgMember[]

  @@index([clerk_org_id])
  @@map("tenants")
}

model OrgMember {
  id            String    @id @default(uuid())
  org_id        String
  tenant_id     String
  clerk_user_id String
  email         String
  full_name     String?
  role          String    @default("viewer")
  is_active     Boolean   @default(true)
  invited_at    DateTime?
  joined_at     DateTime?
  created_at    DateTime  @default(now())
  updated_at    DateTime  @updatedAt

  organization Organization @relation(fields: [org_id], references: [id])
  tenant       Tenant       @relation(fields: [tenant_id], references: [id])
  member_roles MemberRole[]

import { Injectable, OnModuleInit, OnModuleDestroy, Logger } from '@nestjs/common';
import { PrismaClient } from '@prisma/client';

// Modelos que exigem tenant_id em TODA operação
const TENANT_SCOPED_MODELS = new Set([
  'Artist', 'Work', 'Phonogram', 'Contract', 'ContractTemplate',
  'Transaction', 'Invoice', 'Client', 'Lead', 'LeadInteraction',
  'Campaign', 'Briefing', 'Event', 'Project', 'Takedown',
  'Integration', 'SupportTicket', 'Notification', 'Upload', 'AIJob',
]);

@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(PrismaService.name);

  constructor() {
    super({
      log: process.env.NODE_ENV === 'development'
        ? [{ emit: 'event', level: 'query' }]
        : ['error'],
    });
  }

  async onModuleInit() {
    await this.$connect();
    this.logger.log('Conectado ao Neon PostgreSQL');

    if (process.env.NODE_ENV === 'development') {
      (this.$on as any)('query', (e: { duration: number; query: string }) => {
        if (e.duration > 500) {
          this.logger.warn(`Query lenta (${e.duration}ms): ${e.query.slice(0, 150)}`);
        }
      });
    }
  }

  async onModuleDestroy() {
    await this.$disconnect();
  }

  /**
   * OBRIGATÓRIO para toda operação de negócio.
   * Injeta tenant_id automaticamente e filtra deleted_at.
   * Previne IDOR ao nível do ORM.
   */
  forTenant(tenantId: string) {
    return this.$extends({
      query: {
        $allModels: {
          async findMany({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId, deleted_at: null };
            }
            return query(args);
          },
          async findFirst({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId, deleted_at: null };
            }
            return query(args);
          },
          async findUnique({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId };
            }
            return query(args);
          },
          async create({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.data = { ...args.data, tenant_id: tenantId };
            }
            return query(args);
          },
          async update({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              // Força verificação de tenant no WHERE — anti-IDOR
              args.where = { ...args.where, tenant_id: tenantId };
            }
            return query(args);
          },
          async updateMany({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId };
            }
            return query(args);
          },
          async delete({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              // Soft delete automático
              return (query as any)({
                ...args,
                data: { deleted_at: new Date() },
              });
            }
            return query(args);
          },
        },
      },
    });
  }
}
1.4 — EncryptionService AES-256-GCM
typescript// apps/api/src/core/security/encryption.service.ts
import { Injectable, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createCipheriv, createDecipheriv, randomBytes } from 'crypto';

@Injectable()
export class EncryptionService implements OnModuleInit {
  private key: Buffer;
  private readonly algorithm = 'aes-256-gcm';

  constructor(private config: ConfigService) {}

  onModuleInit() {
    const hexKey = this.config.get<string>('ENCRYPTION_KEY')!;
    this.key = Buffer.from(hexKey, 'hex');
    if (this.key.length !== 32) {
      throw new Error('ENCRYPTION_KEY deve representar 32 bytes (64 chars hex)');
    }
  }

  encrypt(plaintext: string): string {
    const iv = randomBytes(16);
    const cipher = createCipheriv(this.algorithm, this.key, iv);
    const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
    const authTag = cipher.getAuthTag();
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted.toString('hex')}`;
  }

  decrypt(ciphertext: string): string {
    const [ivHex, tagHex, encHex] = ciphertext.split(':');
    const decipher = createDecipheriv(
      this.algorithm,
      this.key,
      Buffer.from(ivHex, 'hex'),
    );
    decipher.setAuthTag(Buffer.from(tagHex, 'hex'));
    return decipher.update(Buffer.from(encHex, 'hex')) + decipher.final('utf8');
  }

  encryptNullable(value: string | null | undefined): string | null {
    return value ? this.encrypt(value) : null;
  }

  decryptNullable(value: string | null | undefined): string | null {
    return value ? this.decrypt(value) : null;
  }
}
