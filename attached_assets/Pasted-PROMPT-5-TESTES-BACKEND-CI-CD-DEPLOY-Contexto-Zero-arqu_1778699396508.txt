describe('EncryptionService', () => {
  let service: EncryptionService;

  beforeEach(async () => {
    const module = await Test.createTestingModule({
      providers: [
        EncryptionService,
        { provide: ConfigService, useValue: {
          get: (key: string) => key === 'ENCRYPTION_KEY'
            ? '0'.repeat(64) : undefined,
        }},
      ],
    }).compile();
    service = module.get<EncryptionService>(EncryptionService);
  });

  it('deve criptografar e descriptografar corretamente', () => {
    const original = '123.456.789-09';
    const encrypted = service.encrypt(original);
    expect(encrypted).not.toBe(original);
    expect(encrypted).toContain(':'); // formato iv:tag:cipher
    const decrypted = service.decrypt(encrypted);
    expect(decrypted).toBe(original);
  });

  it('deve retornar null para encryptNullable(null)', () => {
    expect(service.encryptNullable(null)).toBeNull();
    expect(service.encryptNullable(undefined)).toBeNull();
  });

  it('deve retornar null para decryptNullable(null)', () => {
    expect(service.decryptNullable(null)).toBeNull();
  });

  it('duas criptografias do mesmo valor devem ser diferentes (IV aleatório)', () => {
    const enc1 = service.encrypt('test');
    const enc2 = service.encrypt('test');
    expect(enc1).not.toBe(enc2);
    expect(service.decrypt(enc1)).toBe('test');
    expect(service.decrypt(enc2)).toBe('test');
  });
});

describe('ArtistsService', () => {
  let service: ArtistsService;
  let mockDb: any;

  const TENANT_ID = 'tenant-test-uuid';

  beforeEach(async () => {
    // Mock do Drizzle DB
    const mockQuery = {
      select: jest.fn().mockReturnThis(),
      from:   jest.fn().mockReturnThis(),
      where:  jest.fn().mockReturnThis(),
      limit:  jest.fn().mockReturnThis(),
      offset: jest.fn().mockReturnThis(),
      orderBy: jest.fn().mockReturnThis(),
      insert:  jest.fn().mockReturnThis(),
      values:  jest.fn().mockReturnThis(),
      returning: jest.fn().mockResolvedValue([{
        id: 'art-1',
        tenant_id: TENANT_ID,
        nome_artistico: 'Artista Teste',
        tipo: 'solo',
        status: 'ativo',
      }]),
      update: jest.fn().mockReturnThis(),
      set:    jest.fn().mockReturnThis(),
    };
    mockDb = mockQuery;

    const module = await Test.createTestingModule({
      providers: [
        ArtistsService,
        { provide: DRIZZLE_DB, useValue: mockDb },
        { provide: EncryptionService, useValue: {
          encryptNullable: (v: string | null) => v ? `enc:${v}` : null,
          decryptNullable: (v: string | null) => v?.replace('enc:', '') ?? null,
        }},
      ],
    }).compile();

    service = module.get<ArtistsService>(ArtistsService);
  });

  it('deve criptografar CPF ao criar artista', async () => {
    const dto = {
      nome_artistico: 'Teste',
      tipo: 'solo' as const,
      cpf_cnpj: '123.456.789-09',
      email: 'test@test.com',
    };

    await service.create(TENANT_ID, 'user-1', dto);

    const insertCall = mockDb.values.mock.calls[0][0];
    expect(insertCall.cpf_cnpj_encrypted).toBe('enc:123.456.789-09');
    expect(insertCall.cpf_cnpj).toBeUndefined();
    expect(insertCall.email_encrypted).toBe('enc:test@test.com');
    expect(insertCall.email).toBeUndefined();
  });

  it('deve filtrar deleted_at IS NULL na listagem', async () => {
    await service.list(TENANT_ID, { offset: 0, limit: 10 });
    // Verificar que a query inclui filtro de deleted_at
    // (via inspeção dos args passados para where())
    expect(mockDb.where).toHaveBeenCalled();
  });

  it('deve fazer soft delete (deleted_at) em vez de DELETE', async () => {
    mockDb.limit.mockResolvedValueOnce([{ id: 'art-1', tenant_id: TENANT_ID }]);
    await service.softDelete(TENANT_ID, 'art-1');
    const setCall = mockDb.set.mock.calls[0][0];
    expect(setCall).toHaveProperty('deleted_at');
    expect(setCall.deleted_at).toBeInstanceOf(Date);
  });
});

PARTE 4 — DEPLOY STAGING
═══════════════════════════════════════════════

Criar .github/workflows/deploy-staging.yml:

name: Deploy Staging

on:
  push:
    branches: [staging]

jobs:
  deploy:
    name: Deploy → Staging
    runs-on: ubuntu-latest
    environment: staging
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20', cache: 'npm' }
      - run: npm ci && cd apps/api && npm ci

      - name: Build API
        run: cd apps/api && npm run build

      - name: Deploy API → Railway Staging
        run: npx @railway/cli deploy --service=music-os-360-api-staging
        env:
          RAILWAY_TOKEN: ${{ secrets.RAILWAY_TOKEN_STAGING }}

      - name: Run Drizzle Migrations (Staging)
        run: cd apps/api && npx drizzle-kit migrate
        env:
          NEON_DATABASE_DIRECT_URL: ${{ secrets.NEON_STAGING_DIRECT_URL }}

      - name: Deploy Frontend → Vercel Staging
        run: npx vercel --token=${{ secrets.VERCEL_TOKEN }} --scope=${{ secrets.VERCEL_TEAM }}
        env:
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}

      - name: Smoke Tests Staging
        run: |
          sleep 30
          curl -f ${{ secrets.STAGING_API_URL }}/api/v1/health
          echo "✅ Staging OK"

Criar .github/workflows/deploy-production.yml:

name: Deploy Production

on:
  push:
    tags: ['v*.*.*']

jobs:
  deploy:
    name: Deploy → Production
    runs-on: ubuntu-latest
    environment: production
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20', cache: 'npm' }
      - run: npm ci && cd apps/api && npm ci

      - name: Build API
        run: cd apps/api && npm run build

      - name: Deploy API → Railway Production
        run: npx @railway/cli deploy --service=music-os-360-api
        env:
          RAILWAY_TOKEN: ${{ secrets.RAILWAY_TOKEN_PRODUCTION }}

      - name: Run Drizzle Migrations (Production)
        run: cd apps/api && npx drizzle-kit migrate
        env:
          NEON_DATABASE_DIRECT_URL: ${{ secrets.NEON_PRODUCTION_DIRECT_URL }}

      - name: Deploy Frontend → Vercel Production
        run: npx vercel --prod --token=${{ secrets.VERCEL_TOKEN }} --scope=${{ secrets.VERCEL_TEAM }}
        env:
          VERCEL_ORG_ID: ${{ secrets.VERCEL_ORG_ID }}
          VERCEL_PROJECT_ID: ${{ secrets.VERCEL_PROJECT_ID }}

      - name: Smoke Tests Production
        run: |
          sleep 45
          curl -f ${{ secrets.PRODUCTION_API_URL }}/api/v1/health
          echo "✅ Production OK"

      - name: Notificar Sentry de novo release
        run: |
          npx @sentry/cli releases new ${{ github.ref_name }}
          npx @sentry/cli releases finalize ${{ github.ref_name }}
        env:
          SENTRY_AUTH_TOKEN: ${{ secrets.SENTRY_AUTH_TOKEN }}
          SENTRY_ORG: ${{ secrets.SENTRY_ORG }}
          SENTRY_PROJECT: ${{ secrets.SENTRY_PROJECT }}
