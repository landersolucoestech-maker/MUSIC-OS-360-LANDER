import Stripe from 'stripe';
import { Injectable, BadRequestException, Logger } from '@nestjs/common';
import { ConfigService }  from '@nestjs/config';
import { PrismaService }  from '../../core/database/prisma.service';
import { WsGateway }      from '../../core/websocket/ws.gateway';

const PLAN_FEATURES: Record<string, Record<string, boolean>> = {
  starter: {
    moduleArtists:    true,  moduleCatalog:  true,
    moduleContracts:  true,  moduleCrm:      true,
    moduleMarketing:  false, moduleAccounting: false,
    moduleMonitoring: false, aiFeatures:     false,
  },
  professional: {
    moduleArtists:    true,  moduleCatalog:     true,
    moduleContracts:  true,  moduleCrm:         true,
    moduleMarketing:  true,  moduleAccounting:  true,
    moduleMonitoring: true,  aiFeatures:        false,
    moduleRh:         true,  moduleEvents:      true,
  },
  enterprise: {
    moduleArtists:    true,  moduleCatalog:     true,
    moduleContracts:  true,  moduleCrm:         true,
    moduleMarketing:  true,  moduleAccounting:  true,
    moduleMonitoring: true,  aiFeatures:        true,
    moduleRh:         true,  moduleEvents:      true,
    moduleLicensing:  true,  multiTenantAdmin:  true,
    analyticsAdvanced: true, whitelabel:        true,
  },
};

@Injectable()
export class StripeService {
  private readonly stripe: Stripe;
  private readonly logger = new Logger(StripeService.name);

  constructor(
    private config: ConfigService,
    private prisma: PrismaService,
    private ws:     WsGateway,
  ) {
    this.stripe = new Stripe(this.config.get<string>('STRIPE_SECRET_KEY')!, {
      apiVersion: '2024-11-20.acacia',
    });
  }

  async createCheckoutSession(params: {
    orgId:      string;
    tenantId:   string;
    plan:       string;
    successUrl: string;
    cancelUrl:  string;
  }) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { org_id: params.orgId },
    });

    const prices: Record<string, string | undefined> = {
      starter:      this.config.get('STRIPE_PRICE_STARTER'),
      professional: this.config.get('STRIPE_PRICE_PROFESSIONAL'),
      enterprise:   this.config.get('STRIPE_PRICE_ENTERPRISE'),
    };

    const priceId = prices[params.plan];
    if (!priceId) throw new BadRequestException(`Plano inválido: ${params.plan}`);

    const session = await this.stripe.checkout.sessions.create({
      mode:      'subscription',
      customer:  sub?.stripe_customer_id?.startsWith('pending_')
                   ? undefined
                   : sub?.stripe_customer_id,
      line_items: [{ price: priceId, quantity: 1 }],
      metadata:   { tenant_id: params.tenantId, org_id: params.orgId, plan: params.plan },
      success_url: params.successUrl,
      cancel_url:  params.cancelUrl,
    });

    return { url: session.url };
  }

  async createPortalSession(orgId: string, returnUrl: string) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { org_id: orgId },
    });
    if (!sub?.stripe_customer_id || sub.stripe_customer_id.startsWith('pending_')) {
      throw new BadRequestException('Sem assinatura ativa');
    }

    const session = await this.stripe.billingPortal.sessions.create({
      customer:   sub.stripe_customer_id,
      return_url: returnUrl,
    });
    return { url: session.url };
  }

  async handleWebhook(signature: string, payload: Buffer) {
    let event: Stripe.Event;
    try {
      event = this.stripe.webhooks.constructEvent(
        payload,
        signature,
        this.config.get<string>('STRIPE_WEBHOOK_SECRET')!,
      );
    } catch {
      throw new BadRequestException('Assinatura Stripe inválida');
    }

    // Idempotência por event ID
    const existing = await this.prisma.webhookEvent.findUnique({
      where: { external_id: event.id },
    });
    if (existing?.status === 'processed') {
      this.logger.log(`Webhook já processado: ${event.id}`);
      return;
    }

    // Registrar evento
    await this.prisma.webhookEvent.upsert({
      where:  { external_id: event.id },
      create: { provider: 'stripe', event_type: event.type, external_id: event.id, payload: event as any },
      update: {},
    });

    try {
      await this.processEvent(event);
      await this.prisma.webhookEvent.update({
        where: { external_id: event.id },
        data:  { status: 'processed', processed_at: new Date() },
      });
    } catch (err) {
      await this.prisma.webhookEvent.update({
        where: { external_id: event.id },
        data:  { status: 'failed', error: (err as Error).message, retry_count: { increment: 1 } },
      });
      throw err;
    }
  }

  private async processEvent(event: Stripe.Event) {
    switch (event.type) {
      case 'checkout.session.completed':
        await this.onCheckoutCompleted(event.data.object as Stripe.Checkout.Session);
        break;
      case 'customer.subscription.updated':
        await this.onSubscriptionUpdated(event.data.object as Stripe.Subscription);
        break;
      case 'customer.subscription.deleted':
        await this.onSubscriptionCanceled(event.data.object as Stripe.Subscription);
        break;
      case 'invoice.payment_failed':
        await this.onPaymentFailed(event.data.object as Stripe.Invoice);
        break;
    }
  }

  private async onCheckoutCompleted(session: Stripe.Checkout.Session) {
    const { tenant_id, org_id, plan } = session.metadata ?? {};
    if (!tenant_id || !org_id || !plan) return;

    const features = PLAN_FEATURES[plan] ?? PLAN_FEATURES.starter;

    await this.prisma.$transaction(async (tx) => {
      // Atualizar subscription
      await tx.billingSubscription.updateMany({
        where: { org_id },
        data: {
          stripe_customer_id: session.customer as string,
          stripe_sub_id:      session.subscription as string,
          plan,
          status:             'active',
          trial_ends_at:      null,
        },
      });

      // Atualizar features do tenant
      await tx.tenant.updateMany({
        where: { id: tenant_id },
        data:  { plan, features },
      });

      // Atualizar plan da org
      await tx.organization.updateMany({
        where: { id: org_id },
        data:  { plan, billing_status: 'active' },
      });
    });

    // Notificar via WebSocket
    await this.ws.sendToTenant(tenant_id, 'billing:plan_upgraded', { plan });
    this.logger.log(`Plano atualizado: ${org_id} → ${plan}`);
  }

  private async onSubscriptionUpdated(sub: Stripe.Subscription) {
    const tenantId = sub.metadata?.tenant_id;
    const orgId    = sub.metadata?.org_id;
    if (!tenantId || !orgId) return;

    await this.prisma.billingSubscription.updateMany({
      where: { stripe_sub_id: sub.id },
      data: {
        status:             sub.status,
        current_period_end: new Date(sub.current_period_end * 1000),
      },
    });
  }

  private async onSubscriptionCanceled(sub: Stripe.Subscription) {
    const tenantId = sub.metadata?.tenant_id;
    const orgId    = sub.metadata?.org_id;
    if (!tenantId || !orgId) return;

    await this.prisma.$transaction(async (tx) => {
      await tx.billingSubscription.updateMany({
        where: { stripe_sub_id: sub.id },
        data:  { status: 'cancelled' },
      });
      await tx.tenant.updateMany({
        where: { id: tenantId },
        data:  { plan: 'starter', features: PLAN_FEATURES.starter },
      });
    });

    await this.ws.sendToTenant(tenantId, 'billing:cancelled', {});
  }

  private async onPaymentFailed(invoice: Stripe.Invoice) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { stripe_customer_id: invoice.customer as string },
      include: { organization: true },
    });
    if (!sub) return;

    this.logger.warn(`Pagamento falhou: ${sub.org_id}`);
    // Enfileirar email de aviso de falha de pagamento
  }
}

// apps/api/src/modules/billing/billing.controller.ts
@Controller('billing')
@UseGuards(ClerkAuthGuard, TenantGuard)
@ApiBearerAuth()
@ApiTags('Billing')
export class BillingController {
  constructor(private stripe: StripeService) {}

  @Post('checkout')
  checkout(@CurrentTenant() tenant: any, @Body() dto: CreateCheckoutDto) {
    return this.stripe.createCheckoutSession({
      orgId:    tenant.org_id,
      tenantId: tenant.id,
      ...dto,
    });
  }

  @Post('portal')
  portal(@CurrentTenant() tenant: any, @Body('returnUrl') returnUrl: string) {
    return this.stripe.createPortalSession(tenant.org_id, returnUrl);
  }

  @Post('/webhooks/stripe')
  @Public()
  webhook(
    @Headers('stripe-signature') sig: string,
    @Req() req: RawBodyRequest<Request>,
  ) {
    return this.stripe.handleWebhook(sig, req.rawBody as Buffer);
  }
}
