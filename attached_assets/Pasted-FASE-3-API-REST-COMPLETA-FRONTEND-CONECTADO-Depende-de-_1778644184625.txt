import { IsOptional, IsInt, IsString, IsBoolean, Min, Max } from 'class-validator';
import { Transform, Type } from 'class-transformer';

export class PaginationDto {
  @IsOptional() @Type(() => Number) @IsInt() @Min(0)
  offset?: number = 0;

  @IsOptional() @Type(() => Number) @IsInt() @Min(1) @Max(200)
  limit?: number = 50;

  @IsOptional() @IsString()
  orderBy?: string = 'created_at';

  @IsOptional() @Transform(({ value }) => value === 'true')
  ascending?: boolean = false;

  @IsOptional() @IsString()
  search?: string;
}

        data,
        timestamp: new Date().toISOString(),
      })),
    );
  }
}

3.2 — AuditService automático
typescript// apps/api/src/core/audit/audit.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../database/prisma.service';

export interface AuditParams {
  tenantId:  string | null;
  userId:    string | null;
  action:    string;
  entity:    string;
  entityId?: string;
  before?:   unknown;
  after?:    unknown;
  ip?:       string;
  userAgent?: string;
  requestId?: string;
}

@Injectable()
export class AuditService {
  constructor(private prisma: PrismaService) {}

  async log(params: AuditParams): Promise<void> {
    try {
      await this.prisma.auditLog.create({
        data: {
          tenant_id:  params.tenantId,
          user_id:    params.userId,
          action:     params.action,
          entity:     params.entity,
          entity_id:  params.entityId,
          before:     params.before as any,
          after:      params.after as any,
          ip_address: params.ip,
          user_agent: params.userAgent,
          request_id: params.requestId,
        },
      });
    } catch {
      // Auditoria nunca deve quebrar a operação principal
    }
  }
}

// apps/api/src/core/interceptors/audit.interceptor.ts
import {
  Injectable, NestInterceptor, ExecutionContext,
  CallHandler, Logger,
} from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap } from 'rxjs/operators';
import { Reflector } from '@nestjs/core';
import { AuditService } from '../audit/audit.service';

export const AUDIT_KEY = 'audit_action';
export const Audit = (action: string) => SetMetadata(AUDIT_KEY, action);

@Injectable()
export class AuditInterceptor implements NestInterceptor {
  private readonly logger = new Logger(AuditInterceptor.name);

  constructor(
    private reflector: Reflector,
    private auditService: AuditService,
  ) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const action = this.reflector.get<string>(AUDIT_KEY, context.getHandler());
    if (!action) return next.handle();

    const request = context.switchToHttp().getRequest();
    const auth    = request.auth;
    const tenant  = request.tenant;

    return next.handle().pipe(
      tap(async (result) => {
        await this.auditService.log({
          tenantId:  tenant?.id ?? null,
          userId:    auth?.userId ?? null,
          action,
          entity:    action.split('.')[0],
          entityId:  result?.id ?? result?.data?.id,
          after:     result,
          ip:        request.ip,
          userAgent: request.headers['user-agent'],
          requestId: request.headers['x-request-id'],
        });
      }),
    );
  }
}

3.3 — Rate limiting com Upstash
typescript// apps/api/src/core/security/rate-limit.service.ts
import { Injectable, HttpException, HttpStatus } from '@nestjs/common';
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class RateLimitService {
  private readonly limiters = new Map<string, Ratelimit>();

  constructor(private config: ConfigService) {
    const redis = new Redis({
      url:   this.config.get<string>('UPSTASH_REDIS_URL')!,
      token: this.config.get<string>('UPSTASH_REDIS_TOKEN')!,
    });

    const configs = {
      auth:   Ratelimit.slidingWindow(5,   '15 m'),
      api:    Ratelimit.slidingWindow(120, '1 m'),
      ai:     Ratelimit.slidingWindow(20,  '1 h'),
      upload: Ratelimit.slidingWindow(30,  '1 h'),
      webhook: Ratelimit.slidingWindow(100, '1 m'),
    };

    for (const [name, limiter] of Object.entries(configs)) {
      this.limiters.set(name, new Ratelimit({
        redis,
        limiter,
        prefix: `@musicos360/${name}`,
      }));
    }
  }

  async check(
    category: 'auth' | 'api' | 'ai' | 'upload' | 'webhook',
    identifier: string,
  ): Promise<void> {
    const limiter = this.limiters.get(category)!;
    const { success, reset } = await limiter.limit(identifier);

    if (!success) {
      throw new HttpException(
        {
          message:    'Muitas requisições. Tente novamente em breve.',
          retryAfter: Math.ceil((reset - Date.now()) / 1000),
        },
        HttpStatus.TOO_MANY_REQUESTS,
      );
    }
  }
}

// apps/api/src/core/guards/rate-limit.guard.ts
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { RateLimitService } from '../security/rate-limit.service';

@Injectable()
export class RateLimitGuard implements CanActivate {
  constructor(private rateLimitService: RateLimitService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request  = context.switchToHttp().getRequest();
    const tenantId = request.tenant?.id ?? request.ip;
    await this.rateLimitService.check('api', tenantId);
    return true;
  }
}

    @CurrentUser()   user: any,
    @Body()          dto: CreateArtistDto,

    return this.service.create(tenant.id, user.userId, dto);
  }

  @Patch(':id')
  @RequireRole('editor')
  @Audit('artist.updated')

    @CurrentTenant() tenant: any,

typescript// apps/api/src/modules/artists/artists.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService }    from '../../core/database/prisma.service';
import { EncryptionService } from '../../core/security/encryption.service';
import { CreateArtistDto }  from './dto/create-artist.dto';
import { UpdateArtistDto }  from './dto/update-artist.dto';
import { QueryArtistDto }   from './dto/query-artist.dto';

@Injectable()
export class ArtistsService {
  constructor(
    private prisma:     PrismaService,
    private encryption: EncryptionService,
  ) {}

  async list(tenantId: string, query: QueryArtistDto) {
    const db = this.prisma.forTenant(tenantId);

    const where: any = {};
    if (query.status) where.status = query.status;
    if (query.search) {
      where.OR = [
        { nome_artistico: { contains: query.search, mode: 'insensitive' } },
        { genero_musical:  { contains: query.search, mode: 'insensitive' } },
      ];
    }

    const [data, total] = await Promise.all([
      db.artist.findMany({
        where,
        orderBy: { [query.orderBy ?? 'created_at']: query.ascending ? 'asc' : 'desc' },
        skip:    query.offset ?? 0,
        take:    query.limit  ?? 50,
      }),
      db.artist.count({ where }),
    ]);

    return {
      data: data.map(a => this.toResponse(a)),
      meta: { total, offset: query.offset ?? 0, limit: query.limit ?? 50 },
    };
  }

  async findById(tenantId: string, id: string) {
    const db     = this.prisma.forTenant(tenantId);
    const artist = await db.artist.findFirst({ where: { id } });
    if (!artist) throw new NotFoundException('Artista não encontrado');
    return this.toResponse(artist);
  }

  async create(tenantId: string, userId: string, dto: CreateArtistDto) {
    const db     = this.prisma.forTenant(tenantId);
    const artist = await db.artist.create({
      data: {
        ...this.toEncrypted(dto),
        created_by: userId,
      },
    });
    return this.toResponse(artist);
  }

  async update(tenantId: string, userId: string, id: string, dto: UpdateArtistDto) {
    const db = this.prisma.forTenant(tenantId);
    const existing = await db.artist.findFirst({ where: { id } });
    if (!existing) throw new NotFoundException('Artista não encontrado');

    const updated = await db.artist.update({
      where: { id },
      data:  { ...this.toEncrypted(dto), updated_by: userId },
    });
    return this.toResponse(updated);
  }

  async softDelete(tenantId: string, id: string) {
    const db = this.prisma.forTenant(tenantId);
    await db.artist.update({
      where: { id },
      data:  { deleted_at: new Date() },
    });
    return { deleted: true };
  }

  // Criptografar campos sensíveis antes de salvar
  private toEncrypted(dto: any) {
    const { email, telefone, cpf_cnpj, manager_contato, ...rest } = dto;
    return {
      ...rest,
      email_encrypted:           this.encryption.encryptNullable(email),
      telefone_encrypted:        this.encryption.encryptNullable(telefone),
      cpf_cnpj_encrypted:        this.encryption.encryptNullable(cpf_cnpj),
      manager_contato_encrypted: this.encryption.encryptNullable(manager_contato),
    };
  }

  // Descriptografar antes de retornar ao frontend
  private toResponse(artist: any) {
    const {
      email_encrypted,
      telefone_encrypted,
      cpf_cnpj_encrypted,
      manager_contato_encrypted,
      ...rest
    } = artist;
    return {
      ...rest,
      email:           this.encryption.decryptNullable(email_encrypted),
      telefone:        this.encryption.decryptNullable(telefone_encrypted),
      cpf_cnpj:        this.encryption.decryptNullable(cpf_cnpj_encrypted),
      manager_contato: this.encryption.decryptNullable(manager_contato_encrypted),
    };
  }
}
