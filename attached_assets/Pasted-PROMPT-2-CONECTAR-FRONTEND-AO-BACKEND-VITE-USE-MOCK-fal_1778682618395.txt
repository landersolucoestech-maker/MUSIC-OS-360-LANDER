PARTE 2 — ATUALIZAR API-CLIENT PARA ENVIAR TOKEN CLERK
═══════════════════════════════════════════════

Atualizar client/src/shared/lib/api-client.ts.

O token JWT deve ser renovado automaticamente via Clerk.
Adicionar função refreshClerkToken():

import { getToken } from '@clerk/clerk-react';  // NÃO PODE: hook fora de componente

// SOLUÇÃO: usar uma variável global atualizada pelo AuthContext

// No topo do arquivo, já existe:
let _accessToken: string | null = null;
export function setAccessToken(t: string | null): void { _accessToken = t; }
export function getAccessToken(): string | null { return _accessToken; }

// No AuthContext.tsx, adicionar refresh automático do token:
// Quando isSignedIn=true, chamar getToken() e setAccessToken() periodicamente.

Atualizar apps/web/../client/src/app/providers/AuthContext.tsx.
Na função AuthContextBridge, adicionar:

useEffect(() => {
  if (!isSignedIn) {
    setAccessToken(null);
    return;
  }
  
  // Refresh inicial
  const refreshToken = async () => {
    try {
      const token = await getToken();
      if (token) {
        const { setAccessToken } = await import('@/shared/lib/api-client');
        setAccessToken(token);
      }
    } catch (err) {
      console.error('Falha ao renovar token Clerk:', err);
    }
  };
  
  refreshToken();
  
  // Renovar a cada 55 minutos (token Clerk expira em 60min)
  const interval = setInterval(refreshToken, 55 * 60 * 1000);
  return () => clearInterval(interval);
}, [isSignedIn, getToken]);
