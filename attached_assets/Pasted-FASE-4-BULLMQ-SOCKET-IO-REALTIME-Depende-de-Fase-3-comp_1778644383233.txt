import { Processor, WorkerHost } from '@nestjs/bullmq';
import { Job } from 'bullmq';
import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '../../core/database/prisma.service';
import { WsGateway }     from '../../core/websocket/ws.gateway';
import { QUEUES }        from '../../core/queue/queue.constants';

export interface NotificationJobData {
  tenantId:  string;
  userId:    string;
  title:     string;
  body?:     string;
  type:      string;
  entity?:   string;
  entityId?: string;
}

@Processor(QUEUES.NOTIFICATIONS)
@Injectable()
export class NotificationsProcessor extends WorkerHost {
  private readonly logger = new Logger(NotificationsProcessor.name);

  constructor(
    private prisma: PrismaService,
    private ws:     WsGateway,
  ) { super(); }

  async process(job: Job<NotificationJobData>) {
    const d = job.data;

    // 1. Persistir no banco
    const notification = await this.prisma.notification.create({
      data: {
        tenant_id: d.tenantId,
        user_id:   d.userId,
        title:     d.title,
        body:      d.body,
        type:      d.type,
        entity:    d.entity,
        entity_id: d.entityId,
      },
    });

    // 2. Enviar via WebSocket ao usuário
    await this.ws.sendToUser(d.tenantId, d.userId, 'notification:new', {
      id:    notification.id,
      title: notification.title,
      body:  notification.body,
      type:  notification.type,
      created_at: notification.created_at,
    });

    this.logger.log(`Notificação entregue: ${d.userId} — ${d.title}`);
    return notification;
  }
}

import {
  WebSocketGateway, WebSocketServer,
  OnGatewayConnection, OnGatewayDisconnect,
  OnGatewayInit, SubscribeMessage,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { createAdapter }  from '@socket.io/redis-adapter';
import { Redis }          from 'ioredis';
import { Logger }         from '@nestjs/common';
import { createClerkClient } from '@clerk/backend';
import { ConfigService }  from '@nestjs/config';

@WebSocketGateway({
  cors: {
    origin:      process.env.CORS_ORIGINS?.split(',') ?? ['http://localhost:5000'],
    credentials: true,
  },
  transports: ['websocket', 'polling'],
  namespace:  '/realtime',
})
export class WsGateway implements OnGatewayInit, OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  private server: Server;

  private readonly logger = new Logger(WsGateway.name);
  private clerk: ReturnType<typeof createClerkClient>;

  // userId → Set de socketIds (suporte a múltiplas abas)
  private userSockets = new Map<string, Set<string>>();

  constructor(private config: ConfigService) {
    this.clerk = createClerkClient({
      secretKey: config.get<string>('CLERK_SECRET_KEY'),
    });
  }

  async afterInit() {
    // Redis Pub/Sub para sincronizar múltiplas instâncias do backend
    const queueUrl = this.config.get<string>('REDIS_QUEUE_URL')!;
    const pub = new Redis(queueUrl);
    const sub = pub.duplicate();
    this.server.adapter(createAdapter(pub, sub));
    this.logger.log('WebSocket Gateway iniciado com Redis Pub/Sub');
  }

  async handleConnection(socket: Socket) {
    const token = socket.handshake.auth?.token
      ?? socket.handshake.headers['authorization']?.replace('Bearer ', '');

    if (!token) {
      socket.disconnect(true);
      return;
    }

    try {
      const payload = await this.clerk.verifyToken(token);

      socket.data.userId   = payload.sub;
      socket.data.tenantId = payload.org_id;

      // Room do tenant (dados compartilhados da org)
      await socket.join(`tenant:${payload.org_id}`);
      // Room pessoal do usuário (notificações individuais)
      await socket.join(`user:${payload.sub}`);

      if (!this.userSockets.has(payload.sub)) {
        this.userSockets.set(payload.sub, new Set());
      }
      this.userSockets.get(payload.sub)!.add(socket.id);

      this.logger.log(`WS conectado: ${payload.sub} / tenant ${payload.org_id}`);
    } catch {
      socket.disconnect(true);
    }
  }

  handleDisconnect(socket: Socket) {
    const { userId } = socket.data;
    if (userId) {
      const sockets = this.userSockets.get(userId);
      sockets?.delete(socket.id);
      if (sockets?.size === 0) this.userSockets.delete(userId);
    }
  }

  // Enviar para todos os usuários do tenant
  async sendToTenant(tenantId: string, event: string, data: unknown) {
    this.server.to(`tenant:${tenantId}`).emit(event, data);
  }

  // Enviar para um usuário específico (com verificação de tenant)
  async sendToUser(tenantId: string, userId: string, event: string, data: unknown) {
    this.server
      .to(`tenant:${tenantId}`)
      .to(`user:${userId}`)
      .emit(event, data);
  }

  // Notificar todo o tenant que um recurso foi atualizado
  // (frontend invalida a query correspondente)
  async notifyDataChanged(tenantId: string, entity: string, id: string) {
    this.server.to(`tenant:${tenantId}`).emit('data:changed', { entity, id });
  }
}

import {
  S3Client, PutObjectCommand,
  GetObjectCommand, DeleteObjectCommand,
} from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { Injectable, BadRequestException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { randomUUID }    from 'crypto';

const ALLOWED_MIMES = {
  documents:    ['application/pdf', 'application/msword',
                 'application/vnd.openxmlformats-officedocument.wordprocessingml.document'],
  images:       ['image/jpeg', 'image/png', 'image/webp', 'image/gif'],
  audio:        ['audio/mpeg', 'audio/wav', 'audio/ogg', 'audio/flac', 'audio/mp4'],
  spreadsheets: ['application/vnd.ms-excel',
                 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
                 'text/csv'],
} as const;

const MAX_SIZES_MB = {
  documents:    50,
  images:       10,
  audio:        500,
  spreadsheets: 20,
};

type UploadCategory = keyof typeof ALLOWED_MIMES;

@Injectable()
export class R2Service {
  private readonly s3: S3Client;
  private readonly bucket: string;

  constructor(private config: ConfigService) {
    this.bucket = config.get<string>('R2_BUCKET_NAME')!;

    this.s3 = new S3Client({
      region:   'auto',
      endpoint: `https://${config.get('R2_ACCOUNT_ID')}.r2.cloudflarestorage.com`,
      credentials: {
        accessKeyId:     config.get<string>('R2_ACCESS_KEY')!,
        secretAccessKey: config.get<string>('R2_SECRET_KEY')!,
      },
    });
  }

  async createPresignedUpload(params: {
    tenantId:   string;
    userId:     string;
    category:   UploadCategory;
    fileName:   string;
    mimeType:   string;
    sizeBytes:  number;
    entity?:    string;
    entityId?:  string;
  }) {
    // Validações obrigatórias server-side
    if (!ALLOWED_MIMES[params.category].includes(params.mimeType as any)) {
      throw new BadRequestException(`Tipo de arquivo não permitido: ${params.mimeType}`);
    }

    const maxBytes = MAX_SIZES_MB[params.category] * 1024 * 1024;
    if (params.sizeBytes > maxBytes) {
      throw new BadRequestException(
        `Arquivo muito grande. Máximo: ${MAX_SIZES_MB[params.category]}MB`,
      );
    }

    // Path com isolamento de tenant
    const fileId = randomUUID();
    const safeFileName = params.fileName.replace(/[^a-zA-Z0-9._-]/g, '_');
    const key = `tenants/${params.tenantId}/${params.category}/${fileId}/${safeFileName}`;

    const command = new PutObjectCommand({
      Bucket:        this.bucket,
      Key:           key,
      ContentType:   params.mimeType,
      ContentLength: params.sizeBytes,
      Metadata: {
        tenant_id:     params.tenantId,
        user_id:       params.userId,
        original_name: params.fileName,
        entity:        params.entity    ?? '',
        entity_id:     params.entityId  ?? '',
      },
    });

    const presignedUrl = await getSignedUrl(this.s3, command, { expiresIn: 300 });
    return { presignedUrl, key, fileId };
  }

  async createDownloadUrl(key: string, expiresInSeconds = 3600): Promise<string> {
    const command = new GetObjectCommand({ Bucket: this.bucket, Key: key });
    return getSignedUrl(this.s3, command, { expiresIn: expiresInSeconds });
  }

  async delete(key: string): Promise<void> {
    await this.s3.send(new DeleteObjectCommand({ Bucket: this.bucket, Key: key }));
  }
}

// apps/api/src/modules/uploads/uploads.controller.ts
@Controller('uploads')
@UseGuards(ClerkAuthGuard, TenantGuard)
@ApiBearerAuth()
@ApiTags('Uploads')
export class UploadsController {
  constructor(
    private r2:     R2Service,
    private prisma: PrismaService,
  ) {}

  @Post('presign')
  @ApiOperation({ summary: 'Obter URL pré-assinada para upload direto ao R2' })
  async presign(
    @CurrentTenant() tenant: any,
    @CurrentUser()   user: any,
    @Body()          dto: PresignUploadDto,
  ) {
    const { presignedUrl, key, fileId } = await this.r2.createPresignedUpload({
      tenantId:  tenant.id,
      userId:    user.userId,
      ...dto,
    });

    // Registrar upload pendente no banco
    await this.prisma.upload.create({
      data: {
        tenant_id:     tenant.id,
        user_id:       user.userId,
        filename:      fileId,
        original_name: dto.fileName,
        mime_type:     dto.mimeType,
        size_bytes:    BigInt(dto.sizeBytes),
        r2_key:        key,
        entity:        dto.entity,
        entity_id:     dto.entityId,
      },
    });

    return { presignedUrl, key, fileId };
  }

  @Post(':id/confirm')
  @ApiOperation({ summary: 'Confirmar que upload foi concluído' })
  async confirm(
    @CurrentTenant() tenant: any,
    @Param('id')     fileId: string,
  ) {
    const upload = await this.prisma.forTenant(tenant.id).upload.findFirst({
      where: { filename: fileId },
    });
    if (!upload) throw new NotFoundException('Upload não encontrado');

    return this.prisma.upload.update({
      where: { id: upload.id },
      data:  { r2_url: `confirmed:${upload.r2_key}` },
    });
  }

  @Get(':id/download')
  @ApiOperation({ summary: 'Obter URL temporária de download' })
  async download(
    @CurrentTenant() tenant: any,
    @Param('id')     fileId: string,
  ) {
    const upload = await this.prisma.forTenant(tenant.id).upload.findFirst({
      where: { filename: fileId },
    });
    if (!upload) throw new NotFoundException('Arquivo não encontrado');

    const url = await this.r2.createDownloadUrl(upload.r2_key);
    return { url, expiresIn: 3600 };
  }
}

import Stripe from 'stripe';
import { Injectable, BadRequestException, Logger } from '@nestjs/common';
import { ConfigService }  from '@nestjs/config';
import { PrismaService }  from '../../core/database/prisma.service';
import { WsGateway }      from '../../core/websocket/ws.gateway';

const PLAN_FEATURES: Record<string, Record<string, boolean>> = {
  starter: {
    moduleArtists:    true,  moduleCatalog:  true,
    moduleContracts:  true,  moduleCrm:      true,
    moduleMarketing:  false, moduleAccounting: false,
    moduleMonitoring: false, aiFeatures:     false,
  },
  professional: {
    moduleArtists:    true,  moduleCatalog:     true,
    moduleContracts:  true,  moduleCrm:         true,
    moduleMarketing:  true,  moduleAccounting:  true,
    moduleMonitoring: true,  aiFeatures:        false,
    moduleRh:         true,  moduleEvents:      true,
  },
  enterprise: {
    moduleArtists:    true,  moduleCatalog:     true,
    moduleContracts:  true,  moduleCrm:         true,
    moduleMarketing:  true,  moduleAccounting:  true,
    moduleMonitoring: true,  aiFeatures:        true,
    moduleRh:         true,  moduleEvents:      true,
    moduleLicensing:  true,  multiTenantAdmin:  true,
    analyticsAdvanced: true, whitelabel:        true,
  },
};

@Injectable()
export class StripeService {
  private readonly stripe: Stripe;
  private readonly logger = new Logger(StripeService.name);

  constructor(
    private config: ConfigService,
    private prisma: PrismaService,
    private ws:     WsGateway,
  ) {
    this.stripe = new Stripe(this.config.get<string>('STRIPE_SECRET_KEY')!, {
      apiVersion: '2024-11-20.acacia',
    });
  }

  async createCheckoutSession(params: {
    orgId:      string;
    tenantId:   string;
    plan:       string;
    successUrl: string;
    cancelUrl:  string;
  }) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { org_id: params.orgId },
    });

    const prices: Record<string, string | undefined> = {
      starter:      this.config.get('STRIPE_PRICE_STARTER'),
      professional: this.config.get('STRIPE_PRICE_PROFESSIONAL'),
      enterprise:   this.config.get('STRIPE_PRICE_ENTERPRISE'),
    };

    const priceId = prices[params.plan];
    if (!priceId) throw new BadRequestException(`Plano inválido: ${params.plan}`);

    const session = await this.stripe.checkout.sessions.create({
      mode:      'subscription',
      customer:  sub?.stripe_customer_id?.startsWith('pending_')
                   ? undefined
                   : sub?.stripe_customer_id,
      line_items: [{ price: priceId, quantity: 1 }],
      metadata:   { tenant_id: params.tenantId, org_id: params.orgId, plan: params.plan },
      success_url: params.successUrl,
      cancel_url:  params.cancelUrl,
    });

    return { url: session.url };
  }

  async createPortalSession(orgId: string, returnUrl: string) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { org_id: orgId },
    });
    if (!sub?.stripe_customer_id || sub.stripe_customer_id.startsWith('pending_')) {
      throw new BadRequestException('Sem assinatura ativa');
    }

    const session = await this.stripe.billingPortal.sessions.create({
      customer:   sub.stripe_customer_id,
      return_url: returnUrl,
    });
    return { url: session.url };
  }

  async handleWebhook(signature: string, payload: Buffer) {
    let event: Stripe.Event;
    try {
      event = this.stripe.webhooks.constructEvent(
        payload,
        signature,
        this.config.get<string>('STRIPE_WEBHOOK_SECRET')!,
      );
    } catch {
      throw new BadRequestException('Assinatura Stripe inválida');
    }

    // Idempotência por event ID
    const existing = await this.prisma.webhookEvent.findUnique({
      where: { external_id: event.id },
    });
    if (existing?.status === 'processed') {
      this.logger.log(`Webhook já processado: ${event.id}`);
      return;
    }

    // Registrar evento
    await this.prisma.webhookEvent.upsert({
      where:  { external_id: event.id },
      create: { provider: 'stripe', event_type: event.type, external_id: event.id, payload: event as any },
      update: {},
    });

    try {
      await this.processEvent(event);
      await this.prisma.webhookEvent.update({
        where: { external_id: event.id },
        data:  { status: 'processed', processed_at: new Date() },
      });
    } catch (err) {
      await this.prisma.webhookEvent.update({
        where: { external_id: event.id },
        data:  { status: 'failed', error: (err as Error).message, retry_count: { increment: 1 } },
      });
      throw err;
    }
  }

  private async processEvent(event: Stripe.Event) {
    switch (event.type) {
      case 'checkout.session.completed':
        await this.onCheckoutCompleted(event.data.object as Stripe.Checkout.Session);
        break;
      case 'customer.subscription.updated':
        await this.onSubscriptionUpdated(event.data.object as Stripe.Subscription);
        break;
      case 'customer.subscription.deleted':
        await this.onSubscriptionCanceled(event.data.object as Stripe.Subscription);
        break;
      case 'invoice.payment_failed':
        await this.onPaymentFailed(event.data.object as Stripe.Invoice);
        break;
    }
  }

  private async onCheckoutCompleted(session: Stripe.Checkout.Session) {
    const { tenant_id, org_id, plan } = session.metadata ?? {};
    if (!tenant_id || !org_id || !plan) return;

    const features = PLAN_FEATURES[plan] ?? PLAN_FEATURES.starter;

    await this.prisma.$transaction(async (tx) => {
      // Atualizar subscription
      await tx.billingSubscription.updateMany({
        where: { org_id },
        data: {
          stripe_customer_id: session.customer as string,
          stripe_sub_id:      session.subscription as string,
          plan,
          status:             'active',
          trial_ends_at:      null,
        },
      });

      // Atualizar features do tenant
      await tx.tenant.updateMany({
        where: { id: tenant_id },
        data:  { plan, features },
      });

      // Atualizar plan da org
      await tx.organization.updateMany({
        where: { id: org_id },
        data:  { plan, billing_status: 'active' },
      });
    });

    // Notificar via WebSocket
    await this.ws.sendToTenant(tenant_id, 'billing:plan_upgraded', { plan });
    this.logger.log(`Plano atualizado: ${org_id} → ${plan}`);
  }

  private async onSubscriptionUpdated(sub: Stripe.Subscription) {
    const tenantId = sub.metadata?.tenant_id;
    const orgId    = sub.metadata?.org_id;
    if (!tenantId || !orgId) return;

    await this.prisma.billingSubscription.updateMany({
      where: { stripe_sub_id: sub.id },
      data: {
        status:             sub.status,
        current_period_end: new Date(sub.current_period_end * 1000),
      },
    });
  }

  private async onSubscriptionCanceled(sub: Stripe.Subscription) {
    const tenantId = sub.metadata?.tenant_id;
    const orgId    = sub.metadata?.org_id;
    if (!tenantId || !orgId) return;

    await this.prisma.$transaction(async (tx) => {
      await tx.billingSubscription.updateMany({
        where: { stripe_sub_id: sub.id },
        data:  { status: 'cancelled' },
      });
      await tx.tenant.updateMany({
        where: { id: tenantId },
        data:  { plan: 'starter', features: PLAN_FEATURES.starter },
      });
    });

    await this.ws.sendToTenant(tenantId, 'billing:cancelled', {});
  }

  private async onPaymentFailed(invoice: Stripe.Invoice) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { stripe_customer_id: invoice.customer as string },
      include: { organization: true },
    });
    if (!sub) return;

    this.logger.warn(`Pagamento falhou: ${sub.org_id}`);
    // Enfileirar email de aviso de falha de pagamento
  }
}

@Controller('billing')
@UseGuards(ClerkAuthGuard, TenantGuard)
@ApiBearerAuth()
@ApiTags('Billing')
export class BillingController {
  constructor(private stripe: StripeService) {}

  @Post('checkout')
  checkout(@CurrentTenant() tenant: any, @Body() dto: CreateCheckoutDto) {
    return this.stripe.createCheckoutSession({
      orgId:    tenant.org_id,
      tenantId: tenant.id,
      ...dto,
    });
  }

  @Post('portal')
  portal(@CurrentTenant() tenant: any, @Body('returnUrl') returnUrl: string) {
    return this.stripe.createPortalSession(tenant.org_id, returnUrl);
  }

  @Post('/webhooks/stripe')
  @Public()
  webhook(
    @Headers('stripe-signature') sig: string,
    @Req() req: RawBodyRequest<Request>,
  ) {
    return this.stripe.handleWebhook(sig, req.rawBody as Buffer);
  }
}
