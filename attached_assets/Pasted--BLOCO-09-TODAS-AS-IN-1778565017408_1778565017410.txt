NÃO REQUER API BANCÁRIA. Importação manual de ficheiro.

```typescript
@Post("transactions/import-ofx")
@UseInterceptors(FileInterceptor("file"))
async importOFX(@UploadedFile() file: Express.Multer.File, @Req() req) {
  const ofxContent = file.buffer.toString("latin1");
  const ofxParser  = await import("ofx-parser");
  const parsed     = ofxParser.parse(ofxContent);
  const transactions = parsed.OFX.BANKMSGSRSV1.STMTTRNRS.STMTRS.BANKTRANLIST.STMTTRN;

  const mapped = transactions.map(t => ({
    org_id:    req.orgId,
    tipo:      parseFloat(t.TRNAMT) > 0 ? "receita" : "despesa",
    valor:     Math.abs(parseFloat(t.TRNAMT)),
    descricao: t.MEMO,
    data:      parseOFXDate(t.DTPOSTED),
    status:    "pendente",  // aguarda confirmação do utilizador
  }));

  // Retornar para o frontend revisar antes de importar definitivamente
  return { preview: mapped, total: mapped.length };
}
```

```typescript
// Após receber o JWT do backend, actualizar o tenant com o role correcto:

// Adicionar função no AuthContext:
export function useSyncTenantFromJWT() {
  const { tenant, setTenant } = useTenant();
  const { accessToken } = useAuth();
  
  useEffect(() => {
    if (!accessToken || MOCK_MODE) return;
    try {
      const decoded = jwtDecode<JwtPayload>(accessToken);
      setTenant(prev => ({
        ...prev,
        id:          decoded.org_id,
        role:        decoded.role as TenantRole,
        permissions: ROLE_PERMISSIONS[decoded.role as TenantRole] ?? ROLE_PERMISSIONS.viewer,
      }));
    } catch (e) { /* ignore */ }
  }, [accessToken]);
}
```

```typescript
// server/src/modules/auth/auth.service.ts
import { JwtService }  from "@nestjs/jwt";
import { clerkClient } from "@clerk/backend";

@Injectable()
export class AuthService {
  constructor(private jwt: JwtService, private prisma: PrismaService) {}

  async syncClerkUser(clerkToken: string): Promise<AuthResult> {
    // 1. Verificar token com Clerk
    const session = await clerkClient.sessions.verifySession(sessionId, clerkToken);
    const clerkUser = await clerkClient.users.getUser(session.userId);

    // 2. Upsert no banco local
    const user = await this.prisma.user.upsert({
      where:  { clerk_id: clerkUser.id },
      update: { email: clerkUser.emailAddresses[0].emailAddress, name: clerkUser.fullName },
      create: { clerk_id: clerkUser.id, email: clerkUser.emailAddresses[0].emailAddress,
                name: clerkUser.fullName },
    });

    // 3. Obter membership e role do tenant
    const membership = await this.prisma.orgMember.findFirst({
      where: { user_id: user.id },
      include: { org: true },
    });

    // 4. Emitir JWT próprio
    const payload: JwtPayload = {
      sub:     user.id,
      email:   user.email,
      role:    (membership?.role ?? "viewer") as TenantRole,
      org_id:  membership?.org_id ?? "",
      org_slug: membership?.org.slug ?? "",
      plan:    membership?.org.plan ?? "starter",
    };

    const access_token  = this.jwt.sign(payload, { expiresIn: "15m" });
    const refresh_token = this.jwt.sign({ sub: user.id }, { expiresIn: "7d" });

    return { access_token, refresh_token, user, org: membership?.org };
  }
}
```

```typescript
// server/src/infrastructure/storage/r2.service.ts
import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";

@Injectable()
export class R2Service {
  private client: S3Client;

  constructor(private config: ConfigService) {
    this.client = new S3Client({
      region:   "auto",
      endpoint: `https://${config.get("R2_ACCOUNT_ID")}.r2.cloudflarestorage.com`,
      credentials: {
        accessKeyId:     config.get("R2_ACCESS_KEY_ID"),
        secretAccessKey: config.get("R2_SECRET_ACCESS_KEY"),
      },
    });
  }

  async upload(key: string, buffer: Buffer, contentType: string): Promise<string> {
    await this.client.send(new PutObjectCommand({
      Bucket: this.config.get("R2_BUCKET_NAME"),
      Key:    key,
      Body:   buffer,
      ContentType: contentType,
    }));
    return `${this.config.get("R2_PUBLIC_URL")}/${key}`;
  }

  async getSignedUrl(key: string, expiresIn = 3600): Promise<string> {
    return getSignedUrl(this.client, new GetObjectCommand({
      Bucket: this.config.get("R2_BUCKET_NAME"),
      Key:    key,
    }), { expiresIn });
  }
}
```

```typescript
// server/src/modules/websocket/ws.gateway.ts
@WebSocketGateway({ cors: { origin: process.env.FRONTEND_URL, credentials: true } })
export class WsGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer() server: Server;
  private readonly logger = new Logger(WsGateway.name);

  constructor(private jwt: JwtService) {}

  async handleConnection(client: Socket) {
    try {
      const token   = client.handshake.auth.token as string;
      const payload = this.jwt.verify<JwtPayload>(token);
      client.join(`org:${payload.org_id}`);
      client.data = { org_id: payload.org_id, user_id: payload.sub };
      this.logger.log(`Client connected: ${payload.sub} → org:${payload.org_id}`);
    } catch {
      client.disconnect();
    }
  }

  handleDisconnect(client: Socket) {
    this.logger.log(`Client disconnected: ${client.id}`);
  }

  emitToOrg(orgId: string, event: string, data: unknown) {
    this.server.to(`org:${orgId}`).emit(event, data);
  }
}
```

```typescript
    Sentry.captureException(exception);

    const res  = ctx.getResponse<Response>();
```
