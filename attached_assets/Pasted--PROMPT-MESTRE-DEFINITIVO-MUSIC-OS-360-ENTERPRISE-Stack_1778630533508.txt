CONTEXTO ABSOLUTO DO PROJETO
O MUSIC OS 360° é um ERP Musical SaaS multi-tenant atualmente implementado como frontend Vite + React puro, 100% em mock mode (localStorage + dados seed em memória). Possui ~107K linhas TypeScript, 22 módulos de negócio, arquitetura modular sólida, 9 tabelas marcadas como PENDING sem backend, todas as integrações simuladas, zero persistência real, zero autenticação real, zero backend funcional.
O objetivo é transformar isso em plataforma SaaS enterprise real.
DECISÃO ARQUITETURAL IMUTÁVEL:

Frontend: Vite + React permanece como está — estrutura, UX, visual e módulos são preservados integralmente
O frontend tem flag VITE_USE_MOCK=true que mantém tudo funcionando enquanto o backend é construído fase a fase
A transição de mock → real acontece módulo por módulo, nunca tudo de uma vez
NÃO existe migração para Next.js neste escopo

typescriptimport { NestFactory } from '@nestjs/core';
import { ValidationPipe, Logger } from '@nestjs/common';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import helmet from 'helmet';
import compression from 'compression';
import { AppModule } from './app.module';
import { GlobalExceptionFilter } from './core/filters/global-exception.filter';
import { TransformInterceptor } from './core/interceptors/transform.interceptor';
import { LoggingInterceptor } from './core/interceptors/logging.interceptor';

async function bootstrap() {
  const logger = new Logger('Bootstrap');
  
  const app = await NestFactory.create(AppModule, {
    bufferLogs: true,
    logger: ['error', 'warn', 'log', 'debug'],
  });

  // Security
  app.use(helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        styleSrc: ["'self'", "'unsafe-inline'"],
        imgSrc: ["'self'", 'data:', 'https:'],
        scriptSrc: ["'self'"],
      },
    },
    crossOriginEmbedderPolicy: false,
  }));
  
  app.use(compression());

  // CORS — origins restritas via env
  app.enableCors({
    origin: (origin, callback) => {
      const allowed = process.env.CORS_ORIGINS?.split(',') ?? ['http://localhost:5000'];
      if (!origin || allowed.includes(origin)) {
        callback(null, true);
      } else {
        callback(new Error(`CORS: ${origin} não permitido`));
      }
    },
    credentials: true,
    methods: ['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'OPTIONS'],
    allowedHeaders: ['Content-Type', 'Authorization', 'X-Tenant-ID', 'X-Request-ID'],
    exposedHeaders: ['X-Request-ID'],
  });

  // Prefixo global + versioning
  app.setGlobalPrefix('api/v1');

  // Pipes, Filters, Interceptors globais
  app.useGlobalPipes(new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
    transformOptions: { enableImplicitConversion: true },
  }));
  app.useGlobalFilters(new GlobalExceptionFilter());
  app.useGlobalInterceptors(
    new TransformInterceptor(),
    new LoggingInterceptor(),
  );

  // Swagger (apenas em dev/staging)
  if (process.env.NODE_ENV !== 'production') {
    const config = new DocumentBuilder()
      .setTitle('MUSIC OS 360° API')
      .setDescription('Enterprise Music Management SaaS API')
      .setVersion('1.0')
      .addBearerAuth()
      .addServer('http://localhost:3001', 'Development')
      .build();
    const document = SwaggerModule.createDocument(app, config);
    SwaggerModule.setup('docs', app, document, {
      swaggerOptions: { persistAuthorization: true },
    });
  }

  // Graceful Shutdown
  app.enableShutdownHooks();
  process.on('SIGTERM', async () => {
    logger.log('SIGTERM recebido — iniciando graceful shutdown...');
    await app.close();
    process.exit(0);
  });

  const port = process.env.PORT ?? 3001;
  await app.listen(port);
  
  logger.log(`🎵 MUSIC OS 360° API rodando em http://localhost:${port}`);
  logger.log(`📚 Swagger disponível em http://localhost:${port}/docs`);
}

bootstrap().catch((err) => {
  console.error('Falha crítica no bootstrap:', err);
  process.exit(1);
});

typescriptimport { z } from 'zod';

const envSchema = z.object({
  NODE_ENV: z.enum(['development', 'staging', 'production', 'test'])
    .default('development'),
  PORT: z.coerce.number().default(3001),
  
  // Database
  DATABASE_URL: z.string().url('DATABASE_URL inválida'),
  DATABASE_POOL_URL: z.string().url().optional(), // Neon PgBouncer
  
  // Redis
  REDIS_QUEUE_URL: z.string().min(1, 'REDIS_QUEUE_URL obrigatório'),
  REDIS_QUEUE_PASSWORD: z.string().optional(),
  UPSTASH_REDIS_URL: z.string().url().optional(),
  UPSTASH_REDIS_TOKEN: z.string().optional(),
  
  // Auth
  CLERK_SECRET_KEY: z.string().startsWith('sk_', 'CLERK_SECRET_KEY inválida'),
  CLERK_WEBHOOK_SECRET: z.string().startsWith('whsec_'),
  
  // CORS
  CORS_ORIGINS: z.string().default('http://localhost:5000'),
  
  // Encryption
  ENCRYPTION_KEY: z.string().length(64, 'ENCRYPTION_KEY deve ter 64 chars hex (256 bits)'),
  ENCRYPTION_IV_SECRET: z.string().min(32),
  
  // Stripe (opcional em dev)
  STRIPE_SECRET_KEY: z.string().optional(),
  STRIPE_WEBHOOK_SECRET: z.string().optional(),
  
  // Storage
  R2_ACCOUNT_ID: z.string().optional(),
  R2_ACCESS_KEY: z.string().optional(),
  R2_SECRET_KEY: z.string().optional(),
  R2_BUCKET_NAME: z.string().optional(),
  R2_PUBLIC_URL: z.string().optional(),
  
  // AI
  OPENAI_API_KEY: z.string().optional(),
  ANTHROPIC_API_KEY: z.string().optional(),
  GOOGLE_AI_API_KEY: z.string().optional(),
  
  // Resend
  RESEND_API_KEY: z.string().optional(),
  RESEND_FROM_EMAIL: z.string().email().default('noreply@musicos360.com.br'),
  
  // Monitoring
  SENTRY_DSN: z.string().url().optional(),
  POSTHOG_API_KEY: z.string().optional(),
});

export type EnvConfig = z.infer<typeof envSchema>;

export function validateEnv(config: Record<string, unknown>): EnvConfig {
  const result = envSchema.safeParse(config);
  if (!result.success) {
    console.error('❌ Variáveis de ambiente inválidas:');
    result.error.issues.forEach(issue => {
      console.error(`  ${issue.path.join('.')}: ${issue.message}`);
    });
    process.exit(1);
  }
  return result.data;
}

import { Injectable, OnModuleInit, OnModuleDestroy, Logger } from '@nestjs/common';
import { PrismaClient } from '@prisma/client';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class PrismaService extends PrismaClient implements OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(PrismaService.name);

  constructor(private config: ConfigService) {
    super({
      datasources: {
        db: { url: config.get('DATABASE_URL') },
      },
      log: [
        { emit: 'event', level: 'query' },
        { emit: 'event', level: 'error' },
        { emit: 'event', level: 'warn' },
      ],
    });
  }

  async onModuleInit() {
    await this.$connect();
    this.logger.log('PostgreSQL conectado via Prisma');

    // Log queries lentas (>1s) em produção
    (this.$on as any)('query', (e: { duration: number; query: string }) => {
      if (e.duration > 1000) {
        this.logger.warn(`Query lenta (${e.duration}ms): ${e.query.substring(0, 200)}`);
      }
    });
  }

  async onModuleDestroy() {
    await this.$disconnect();
  }

  /**
   * Retorna uma instância do Prisma com filtro automático de tenant.
   * TODA operação em tabela tenant-scoped deve usar este método.
   * Nunca passar tenantId direto nos where sem usar este método.
   */
  forTenant(tenantId: string): PrismaClient {
    return this.$extends({
      query: {
        $allModels: {
          async findMany({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId, deleted_at: null };
            }
            return query(args);
          },
          async findFirst({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId, deleted_at: null };
            }
            return query(args);
          },
          async findUnique({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.where = { ...args.where, tenant_id: tenantId };
            }
            return query(args);
          },
          async create({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              args.data = { ...args.data, tenant_id: tenantId };
            }
            return query(args);
          },
          async update({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              // Anti-IDOR: verifica que o registro pertence ao tenant
              args.where = { ...args.where, tenant_id: tenantId };
            }
            return query(args);
          },
          async delete({ args, query, model }: any) {
            if (TENANT_SCOPED_MODELS.has(model)) {
              // Soft delete automático
              return (this as any).update({
                args: { where: args.where, data: { deleted_at: new Date() } },
                query,
                model,
              });
            }
            return query(args);
          },
        },
      },
    }) as unknown as PrismaClient;
  }
}

import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createCipheriv, createDecipheriv, randomBytes, scryptSync } from 'crypto';

@Injectable()
export class EncryptionService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly key: Buffer;

  constructor(private config: ConfigService) {
    const rawKey = config.get<string>('ENCRYPTION_KEY')!;
    // Derivar chave de 32 bytes a partir da env var
    this.key = Buffer.from(rawKey, 'hex').slice(0, 32);
  }

  encrypt(plaintext: string): string {
    const iv = randomBytes(16);
    const cipher = createCipheriv(this.algorithm, this.key, iv);
    const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
    const authTag = cipher.getAuthTag();
    // Formato: iv:authTag:ciphertext (tudo hex)
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted.toString('hex')}`;
  }

  decrypt(ciphertext: string): string {
    const [ivHex, authTagHex, encryptedHex] = ciphertext.split(':');
    const iv = Buffer.from(ivHex, 'hex');
    const authTag = Buffer.from(authTagHex, 'hex');
    const encrypted = Buffer.from(encryptedHex, 'hex');
    const decipher = createDecipheriv(this.algorithm, this.key, iv);
    decipher.setAuthTag(authTag);
    return decipher.update(encrypted) + decipher.final('utf8');
  }

  // Para campos que podem ser null
  encryptNullable(value: string | null | undefined): string | null {
    if (!value) return null;
    return this.encrypt(value);
  }

  decryptNullable(value: string | null | undefined): string | null {
    if (!value) return null;
    return this.decrypt(value);
  }
}

typescript// apps/api/src/modules/auth/clerk.guard.ts
import {
  Injectable, CanActivate, ExecutionContext,
  UnauthorizedException, Logger,
} from '@nestjs/common';
import { createClerkClient } from '@clerk/backend';
import { ConfigService } from '@nestjs/config';
import { Reflector } from '@nestjs/core';
import { IS_PUBLIC_KEY } from '../decorators/public.decorator';

@Injectable()
export class ClerkAuthGuard implements CanActivate {
  private readonly clerk;
  private readonly logger = new Logger(ClerkAuthGuard.name);

  constructor(
    private config: ConfigService,
    private reflector: Reflector,
  ) {
    this.clerk = createClerkClient({
      secretKey: config.get<string>('CLERK_SECRET_KEY'),
    });
  }

  async canActivate(context: ExecutionContext): Promise<boolean> {
    // Rotas marcadas com @Public() não precisam de auth
    const isPublic = this.reflector.getAllAndOverride<boolean>(IS_PUBLIC_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);
    if (isPublic) return true;

    const request = context.switchToHttp().getRequest();
    const token = this.extractToken(request);

    if (!token) {
      throw new UnauthorizedException('Token de autenticação ausente');
    }

    try {
      // Validação real do JWT Clerk — verifica assinatura, expiração, etc.
      const payload = await this.clerk.verifyToken(token);
      
      // Injeta dados no request para uso nos controllers
      request.auth = {
        userId: payload.sub,
        sessionId: payload.sid,
        orgId: payload.org_id,
        orgRole: payload.org_role,
        claims: payload,
      };

      return true;
    } catch (error) {
      this.logger.warn(`JWT inválido: ${error.message}`);
      throw new UnauthorizedException('Token inválido ou expirado');
    }
  }

  private extractToken(request: any): string | null {
    const auth = request.headers['authorization'];
    if (auth?.startsWith('Bearer ')) return auth.substring(7);
    return request.cookies?.['__session'] ?? null;
  }
}

// apps/api/src/modules/auth/tenant.guard.ts
@Injectable()
export class TenantGuard implements CanActivate {
  constructor(private prisma: PrismaService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const auth = request.auth;
    
    if (!auth?.orgId) {
      throw new UnauthorizedException('Organização não identificada no token');
    }

    // Busca o tenant pelo clerk_org_id
    const tenant = await this.prisma.tenant.findUnique({
      where: { clerk_org_id: auth.orgId },
      include: { organization: true },
    });

    if (!tenant || tenant.deleted_at) {
      throw new UnauthorizedException('Tenant não encontrado ou inativo');
    }

    // Verifica se o usuário é membro ativo
    const member = await this.prisma.orgMember.findUnique({
      where: {
        tenant_id_clerk_user_id: {
          tenant_id: tenant.id,
          clerk_user_id: auth.userId,
        },
      },
    });

    if (!member || !member.is_active) {
      throw new ForbiddenException('Usuário não é membro ativo deste tenant');
    }

    // Injeta tenant e member no request
    request.tenant = tenant;
    request.currentMember = member;

    return true;
  }
}

typescript// apps/api/src/modules/auth/webhooks/clerk.webhook.ts
import { Controller, Post, Headers, Body, RawBodyRequest, Req } from '@nestjs/common';
import { Webhook } from 'svix';
import { ConfigService } from '@nestjs/config';

@Controller('webhooks/clerk')
export class ClerkWebhookController {
  constructor(
    private config: ConfigService,
    private clerkSyncService: ClerkSyncService,
  ) {}

  @Post()
  @Public()
  async handleWebhook(
    @Headers('svix-id') svixId: string,
    @Headers('svix-timestamp') svixTimestamp: string,
    @Headers('svix-signature') svixSignature: string,
    @Req() req: RawBodyRequest<Request>,
  ) {
    const secret = this.config.get<string>('CLERK_WEBHOOK_SECRET')!;
    const wh = new Webhook(secret);

    let payload: ClerkWebhookEvent;
    try {
      // Valida assinatura HMAC do Clerk
      payload = wh.verify(req.rawBody, {
        'svix-id': svixId,
        'svix-timestamp': svixTimestamp,
        'svix-signature': svixSignature,
      }) as ClerkWebhookEvent;
    } catch {
      throw new BadRequestException('Assinatura webhook inválida');
    }

    // Processar evento de forma assíncrona via BullMQ
    await this.clerkSyncService.enqueueSync(payload);

    return { received: true };
  }
}

// ClerkSyncService — processa eventos de forma idempotente
@Injectable()
export class ClerkSyncService {
  constructor(
    private prisma: PrismaService,
    @InjectQueue('clerk-sync') private queue: Queue,
  ) {}

  async enqueueSync(event: ClerkWebhookEvent) {
    await this.queue.add(event.type, event, {
      jobId: event.data.id, // idempotência por ID do evento
      attempts: 3,
      backoff: { type: 'exponential', delay: 2000 },
    });
  }

  // Processado pelo ClerkSyncProcessor
  async processOrganizationCreated(data: ClerkOrganization) {
    await this.prisma.$transaction(async (tx) => {
      const org = await tx.organization.upsert({
        where: { clerk_org_id: data.id },
        create: {
          clerk_org_id: data.id,
          name: data.name,
          slug: data.slug,
          industry: 'gravadora',
        },
        update: { name: data.name, slug: data.slug },
      });

      // Criar tenant vinculado à organização
      await tx.tenant.upsert({
        where: { clerk_org_id: data.id },
        create: {
          org_id: org.id,
          clerk_org_id: data.id,
          name: data.name,
          slug: data.slug,
          plan: 'starter',
          features: DEFAULT_FEATURE_FLAGS,
        },
        update: {},
      });

      // Criar roles sistema para o tenant
      await this.seedSystemRoles(tx, org.id);
    });
  }

  async processUserCreated(data: ClerkUser) {
    // Sincronizar membro com as organizações que ele pertence
    for (const membership of data.organization_memberships ?? []) {
      const tenant = await this.prisma.tenant.findUnique({
        where: { clerk_org_id: membership.organization.id },
      });
      if (!tenant) continue;

      await this.prisma.orgMember.upsert({
        where: {
          tenant_id_clerk_user_id: {
            tenant_id: tenant.id,
            clerk_user_id: data.id,
          },
        },
        create: {
          org_id: tenant.org_id,
          tenant_id: tenant.id,
          clerk_user_id: data.id,
          email: data.email_addresses[0]?.email_address ?? '',
          full_name: `${data.first_name ?? ''} ${data.last_name ?? ''}`.trim(),
          role: membership.role === 'org:admin' ? 'admin' : 'viewer',
          joined_at: new Date(),
        },
        update: {
          email: data.email_addresses[0]?.email_address ?? '',
          full_name: `${data.first_name ?? ''} ${data.last_name ?? ''}`.trim(),
        },
      });
    }
  }
}

@Injectable()
export class ArtistsService {
  constructor(
    private prisma: PrismaService,
    private encryption: EncryptionService,
    private audit: AuditService,
  ) {}

  async list(tenantId: string, query: QueryArtistsDto) {
    const db = this.prisma.forTenant(tenantId);
    
    const [data, total] = await Promise.all([
      db.artist.findMany({
        where: {
          ...(query.status && { status: query.status }),
          ...(query.search && {
            OR: [
              { nome_artistico: { contains: query.search, mode: 'insensitive' } },
              { genero_musical: { contains: query.search, mode: 'insensitive' } },
            ],
          }),
        },
        orderBy: { [query.orderBy ?? 'created_at']: query.ascending ? 'asc' : 'desc' },
        skip: query.offset ?? 0,
        take: query.limit ?? 50,
      }),
      db.artist.count({ where: { status: query.status } }),
    ]);

    // Descriptografar campos sensíveis antes de retornar
    return {
      data: data.map(artist => this.decrypt(artist)),
      meta: { total, page: Math.floor((query.offset ?? 0) / (query.limit ?? 50)) + 1 },
    };
  }

  async create(tenantId: string, userId: string, dto: CreateArtistDto) {
    const db = this.prisma.forTenant(tenantId);
    
    const artist = await db.artist.create({
      data: {
        ...dto,
        // Criptografar campos sensíveis
        email_encrypted: this.encryption.encryptNullable(dto.email),
        telefone_encrypted: this.encryption.encryptNullable(dto.telefone),
        cpf_cnpj_encrypted: this.encryption.encryptNullable(dto.cpf_cnpj),
        email: undefined,    // nunca persistir em claro
        telefone: undefined,
        cpf_cnpj: undefined,
        created_by: userId,
      },
    });

    return this.decrypt(artist);
  }

  private decrypt(artist: Artist) {
    return {
      ...artist,
      email: this.encryption.decryptNullable(artist.email_encrypted),
      telefone: this.encryption.decryptNullable(artist.telefone_encrypted),
      cpf_cnpj: this.encryption.decryptNullable(artist.cpf_cnpj_encrypted),
      // Remover campos criptografados brutos da resposta
      email_encrypted: undefined,
      telefone_encrypted: undefined,
      cpf_cnpj_encrypted: undefined,
    };
  }
}

3.3 — Rate Limiting com Upstash
typescript// apps/api/src/core/security/rate-limit.service.ts
import { Injectable, HttpException, HttpStatus } from '@nestjs/common';
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class RateLimitService {
  private readonly limiters: Map<string, Ratelimit> = new Map();

  constructor(private config: ConfigService) {
    const redis = new Redis({
      url: config.get('UPSTASH_REDIS_URL'),
      token: config.get('UPSTASH_REDIS_TOKEN'),
    });

    // Limites por categoria
    this.limiters.set('auth', new Ratelimit({
      redis,
      limiter: Ratelimit.slidingWindow(5, '15m'), // 5 tentativas em 15min
      prefix: '@musicos360/auth',
    }));

    this.limiters.set('api', new Ratelimit({
      redis,
      limiter: Ratelimit.slidingWindow(100, '1m'), // 100 req/min por tenant
      prefix: '@musicos360/api',
    }));

    this.limiters.set('ai', new Ratelimit({
      redis,
      limiter: Ratelimit.slidingWindow(20, '1h'), // 20 req AI/hora por tenant
      prefix: '@musicos360/ai',
    }));

    this.limiters.set('upload', new Ratelimit({
      redis,
      limiter: Ratelimit.slidingWindow(30, '1h'), // 30 uploads/hora por tenant
      prefix: '@musicos360/upload',
    }));
  }

  async check(category: 'auth' | 'api' | 'ai' | 'upload', identifier: string): Promise<void> {
    const limiter = this.limiters.get(category)!;
    const result = await limiter.limit(identifier);

    if (!result.success) {
      throw new HttpException(
        {
          message: 'Muitas requisições. Tente novamente em breve.',
          retryAfter: Math.ceil((result.reset - Date.now()) / 1000),
        },
        HttpStatus.TOO_MANY_REQUESTS,
      );
    }
  }
}

  constructor(
    private prisma: PrismaService,
    private wsGateway: WsGateway,
    private emailService: EmailService,
  ) { super(); }

  async process(job: Job<NotificationJobData>) {
    const { tenantId, userId, title, body, type, entityId } = job.data;

    // 1. Persistir notificação no banco
    const notification = await this.prisma.notification.create({
      data: { tenant_id: tenantId, user_id: userId, title, body, type, entity_id: entityId },
    });

    // 2. Enviar via WebSocket para o usuário (se online)
    await this.wsGateway.sendToUser(tenantId, userId, 'notification', notification);

    // 3. Se crítico, enviar email também
    if (['contract_expiring', 'payment_failed', 'security_alert'].includes(type)) {
      await this.emailService.sendNotificationEmail(userId, notification);
    }

    return notification;
  }
}

import {
  WebSocketGateway, WebSocketServer, SubscribeMessage,
  OnGatewayConnection, OnGatewayDisconnect, ConnectedSocket,
  MessageBody,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { createAdapter } from '@socket.io/redis-adapter';
import { createClient } from 'redis';

@WebSocketGateway({
  cors: {
    origin: process.env.CORS_ORIGINS?.split(',') ?? ['http://localhost:5000'],
    credentials: true,
  },
  transports: ['websocket', 'polling'],
  namespace: '/realtime',
})
export class WsGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer() server: Server;
  
  private readonly logger = new Logger(WsGateway.name);
  // Map de userId → Set de socketIds (um user pode ter múltiplas abas)
  private userSockets = new Map<string, Set<string>>();

  async afterInit() {
    // Configurar Redis Pub/Sub para sincronizar múltiplas instâncias do servidor
    const pubClient = createClient({ url: process.env.REDIS_QUEUE_URL });
    const subClient = pubClient.duplicate();
    await Promise.all([pubClient.connect(), subClient.connect()]);
    this.server.adapter(createAdapter(pubClient, subClient));
    this.logger.log('Socket.IO configurado com Redis Pub/Sub');
  }

  async handleConnection(socket: Socket) {
    const token = socket.handshake.auth?.token;
    
    if (!token) {
      socket.disconnect(true);
      return;
    }

    try {
      // Validar token Clerk
      const payload = await this.clerkClient.verifyToken(token);
      socket.data.userId = payload.sub;
      socket.data.tenantId = payload.org_id;
      
      // Entrar na room do tenant (isolamento multi-tenant)
      await socket.join(`tenant:${payload.org_id}`);
      // Entrar na room pessoal do usuário
      await socket.join(`user:${payload.sub}`);

      // Registrar socket
      if (!this.userSockets.has(payload.sub)) {
        this.userSockets.set(payload.sub, new Set());
      }
      this.userSockets.get(payload.sub)!.add(socket.id);

      this.logger.log(`WebSocket conectado: user ${payload.sub} tenant ${payload.org_id}`);
    } catch {
      socket.disconnect(true);
    }
  }

  handleDisconnect(socket: Socket) {
    const userId = socket.data.userId;
    if (userId) {
      this.userSockets.get(userId)?.delete(socket.id);
    }
  }

  // Enviar evento para todos os usuários de um tenant
  async sendToTenant(tenantId: string, event: string, data: unknown) {
    this.server.to(`tenant:${tenantId}`).emit(event, data);
  }

  // Enviar evento para um usuário específico
  async sendToUser(tenantId: string, userId: string, event: string, data: unknown) {
    // Dupla proteção: room de tenant + room de user
    this.server.to(`tenant:${tenantId}`).to(`user:${userId}`).emit(event, data);
  }
}

import { S3Client, PutObjectCommand, GetObjectCommand, DeleteObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import { Injectable, BadRequestException } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { randomUUID } from 'crypto';

const ALLOWED_MIME_TYPES = {
  documents: ['application/pdf', 'application/msword',
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document'],
  images: ['image/jpeg', 'image/png', 'image/webp', 'image/gif'],
  audio: ['audio/mpeg', 'audio/wav', 'audio/ogg', 'audio/flac', 'audio/mp4'],
  spreadsheets: ['application/vnd.ms-excel',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', 'text/csv'],
};

const MAX_FILE_SIZES = {
  documents: 50 * 1024 * 1024,  // 50MB
  images: 10 * 1024 * 1024,     // 10MB
  audio: 500 * 1024 * 1024,     // 500MB
  spreadsheets: 20 * 1024 * 1024, // 20MB
};

@Injectable()
export class R2Service {
  private readonly s3: S3Client;
  private readonly bucket: string;
  private readonly publicUrl: string;

  constructor(private config: ConfigService) {
    this.bucket = config.get<string>('R2_BUCKET_NAME')!;
    this.publicUrl = config.get<string>('R2_PUBLIC_URL')!;
    
    this.s3 = new S3Client({
      region: 'auto',
      endpoint: `https://${config.get('R2_ACCOUNT_ID')}.r2.cloudflarestorage.com`,
      credentials: {
        accessKeyId: config.get<string>('R2_ACCESS_KEY')!,
        secretAccessKey: config.get<string>('R2_SECRET_KEY')!,
      },
    });
  }

  // Gera URL pré-assinada para upload direto do browser → R2
  // O backend NUNCA recebe o arquivo, só cria a URL temporária
  async generatePresignedUploadUrl(params: {
    tenantId: string;
    userId: string;
    category: keyof typeof ALLOWED_MIME_TYPES;
    fileName: string;
    mimeType: string;
    sizeBytes: number;
  }) {
    // Validações server-side ANTES de gerar a URL
    if (!ALLOWED_MIME_TYPES[params.category].includes(params.mimeType)) {
      throw new BadRequestException(`Tipo de arquivo não permitido: ${params.mimeType}`);
    }
    
    if (params.sizeBytes > MAX_FILE_SIZES[params.category]) {
      throw new BadRequestException(
        `Arquivo muito grande. Máximo: ${MAX_FILE_SIZES[params.category] / 1024 / 1024}MB`
      );
    }

    // Chave com isolamento por tenant: tenant/{tenantId}/{category}/{uuid}/{fileName}
    const fileId = randomUUID();
    const key = `tenant/${params.tenantId}/${params.category}/${fileId}/${params.fileName}`;

    const command = new PutObjectCommand({
      Bucket: this.bucket,
      Key: key,
      ContentType: params.mimeType,
      ContentLength: params.sizeBytes,
      Metadata: {
        tenant_id: params.tenantId,
        user_id: params.userId,
        original_name: params.fileName,
      },
      // Sem ACL público — todos os arquivos são privados por padrão
    });

    const presignedUrl = await getSignedUrl(this.s3, command, { expiresIn: 300 }); // 5min

    return { presignedUrl, key, fileId };
  }

  // Gera URL temporária para download (acesso autenticado)
  async generateDownloadUrl(key: string, expiresInSeconds = 3600) {
    const command = new GetObjectCommand({ Bucket: this.bucket, Key: key });
    return getSignedUrl(this.s3, command, { expiresIn: expiresInSeconds });
  }

  async deleteFile(key: string) {
    await this.s3.send(new DeleteObjectCommand({ Bucket: this.bucket, Key: key }));
  }
}

// uploads.controller.ts — endpoint para o frontend solicitar URL de upload
@Post('presign')
@UseGuards(ClerkAuthGuard, TenantGuard)

  @CurrentTenant() tenant: Tenant,

      r2_key: key,
      entity: dto.entity,
      entity_id: dto.entityId,
    },
  });

  return { presignedUrl, key, fileId };
}

@Injectable()
export class StripeService {
  private readonly stripe: Stripe;

  constructor(private config: ConfigService, private prisma: PrismaService) {
    this.stripe = new Stripe(config.get<string>('STRIPE_SECRET_KEY')!, {
      apiVersion: '2024-11-20.acacia',
    });
  }

  async createCheckoutSession(params: {
    orgId: string;
    tenantId: string;
    planId: string;
    successUrl: string;
    cancelUrl: string;
  }) {
    const sub = await this.prisma.billingSubscription.findFirst({
      where: { org_id: params.orgId },
    });

    const prices = {
      starter: process.env.STRIPE_PRICE_STARTER,
      professional: process.env.STRIPE_PRICE_PROFESSIONAL,
      enterprise: process.env.STRIPE_PRICE_ENTERPRISE,
    };

    return this.stripe.checkout.sessions.create({
      mode: 'subscription',
      customer: sub?.stripe_customer_id,
      line_items: [{ price: prices[params.planId], quantity: 1 }],
      metadata: { tenant_id: params.tenantId, org_id: params.orgId },
      success_url: params.successUrl,
      cancel_url: params.cancelUrl,
      subscription_data: {
        metadata: { tenant_id: params.tenantId },
        trial_period_days: sub?.status === 'trial' ? undefined : 14,
      },
    });
  }

  async handleWebhook(signature: string, payload: Buffer) {
    let event: Stripe.Event;
    try {
      event = this.stripe.webhooks.constructEvent(
        payload,
        signature,
        this.config.get<string>('STRIPE_WEBHOOK_SECRET')!,
      );
    } catch {
      throw new BadRequestException('Assinatura webhook Stripe inválida');
    }

    // Idempotência: verificar se evento já foi processado
    const existing = await this.prisma.webhookEvent.findUnique({
      where: { external_id: event.id },
    });
    if (existing?.status === 'processed') return;

    await this.processEvent(event);
  }

  private async processEvent(event: Stripe.Event) {
    switch (event.type) {
      case 'checkout.session.completed':
        await this.handleCheckoutCompleted(event.data.object as Stripe.Checkout.Session);
        break;
      case 'customer.subscription.updated':
        await this.handleSubscriptionUpdated(event.data.object as Stripe.Subscription);
        break;
      case 'customer.subscription.deleted':
        await this.handleSubscriptionCanceled(event.data.object as Stripe.Subscription);
        break;
      case 'invoice.payment_failed':
        await this.handlePaymentFailed(event.data.object as Stripe.Invoice);
        break;
    }
  }
}

@Injectable()
export class AIService {
  constructor(
    private prisma: PrismaService,
    private rateLimitService: RateLimitService,
    @InjectQueue(QUEUES.AI_JOBS) private aiQueue: Queue,
  ) {}

  async generate(params: {
    tenantId: string;
    userId: string;
    provider: 'openai' | 'claude' | 'gemini';
    model: string;
    prompt: string;
    skill: string;
    stream?: boolean;
  }) {
    // Rate limiting por tenant para AI
    await this.rateLimitService.check('ai', params.tenantId);

    // Jobs longos → fila; jobs curtos → direto
    if (params.stream) {
      return this.generateDirect(params);
    }

    const job = await this.aiQueue.add('generate', params, {
      timeout: 60000,
      attempts: 2,
    });

    return { jobId: job.id, status: 'queued' };
  }

  private async generateDirect(params: AIGenerateParams) {
    const startMs = Date.now();
    let result: string;
    let usage = { input_tokens: 0, output_tokens: 0 };

    if (params.provider === 'openai') {
      const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
      const completion = await openai.chat.completions.create({
        model: params.model,
        messages: [
          { role: 'system', content: SYSTEM_PROMPT_PT_BR },
          { role: 'user', content: params.prompt },
        ],
        max_tokens: 2048,
      });
      result = completion.choices[0]?.message?.content ?? '';
      usage.input_tokens = completion.usage?.prompt_tokens ?? 0;
      usage.output_tokens = completion.usage?.completion_tokens ?? 0;
      
    } else if (params.provider === 'claude') {
      const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
      const message = await anthropic.messages.create({
        model: params.model,
        max_tokens: 2048,
        messages: [{ role: 'user', content: params.prompt }],
        system: SYSTEM_PROMPT_PT_BR,
      });
      result = message.content[0]?.type === 'text' ? message.content[0].text : '';
      usage.input_tokens = message.usage.input_tokens;
      usage.output_tokens = message.usage.output_tokens;
    }

    // Registrar job no banco para cost tracking e auditoria
    await this.prisma.aIJob.create({
      data: {
        tenant_id: params.tenantId,
        user_id: params.userId,
        provider: params.provider,
        model: params.model,
        skill: params.skill,
        prompt: params.prompt,
        result,
        status: 'completed',
        input_tokens: usage.input_tokens,
        output_tokens: usage.output_tokens,
        cost_usd: this.calculateCost(params.provider, params.model, usage),
        latency_ms: Date.now() - startMs,
        completed_at: new Date(),
      },
    });

    return { content: result, usage };
  }
}
